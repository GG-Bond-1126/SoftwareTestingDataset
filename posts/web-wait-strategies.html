from playwright.sync_api import sync_playwright, expect

项目进阶：PO + 数据驱动 + CI/CD

python — PO 框架
class LoginPage:
    URL = "https://the-internet.herokuapp.com/login"
    def __init__(self, page):
        self.page = page
    def open(self):
        self.page.goto(self.URL)
    def login(self, u, p):
        self.page.fill("#username", u)
        self.page.fill("#password", p)
        self.page.click("button[type=submit]")
    def flash(self):
        return self.page.locator(".flash").inner_text()

import pytest
@pytest.mark.parametrize("u, p, msg", [("tomorrow","secret","You logged in")])
def test_login(u, p, msg):
    from playwright.sync_api import sync_playwright
    with sync_playwright() as pl:
        b = pl.chromium.launch()
        pg = b.new_page()
        lp = LoginPage(pg)
        lp.open(); lp.login(u, p)
        assert msg in lp.flash()
        b.close()

为什么 time.sleep 是「定时炸弹」

python — 反例
import time

driver.get("https://app.example.com")
time.sleep(3)   # ← 定时炸弹
driver.find_element(By.ID, "result").click()

# 问题：
# 1. 网络慢 → 3 秒不够 → 用例挂
# 2. 网络快 → 3 秒太久 → 用例浪费时间（200 个用例 * 3s = 10 分钟）
# 3. 不同环境（CI vs 本地）耗时不同 → 同一套代码在不同机器表现不同
# 4. 页面有"加载动画" → 元素已渲染但还在动画 → 点击无效

time.sleep 的唯一合理用法：等待「固定时间才能完成」的事（如等倒计时结束）。其他场景（等元素出现、等 AJAX 完成）都用显式等待。

显式等待（WebDriverWait）

显式等待 = 「等条件满足，最多等 N 秒」。Selenium 的 WebDriverWait + expected_conditions 是核心。

python — 5 种常用 expected_conditions
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.common.by import By

driver = webdriver.Chrome()
wait = WebDriverWait(driver, 10)   # 默认等 10s，可配置

# 1. 元素出现（存在，不一定可见）
el = wait.until(EC.presence_of_element_located((By.ID, "result")))

# 2. 元素可见（存在 + 可见，可交互）
el = wait.until(EC.visibility_of_element_located((By.ID, "result")))

# 3. 元素可点击（可见 + enabled，最常用）
el = wait.until(EC.element_to_be_clickable((By.ID, "btn")))

# 4. 元素消失（反例：等"加载提示"消失）
wait.until(EC.invisibility_of_element_located((By.ID, "loading")))

# 5. 元素存在（多个匹配，至少 1 个）
elements = wait.until(EC.presence_of_all_elements_located((By.CSS_SELECTOR, ".item")))

# 6. 文字出现（等页面包含某文字）
wait.until(EC.text_to_be_present_in_element((By.ID, "msg"), "成功"))

# 7. URL 变化（等跳转完成）
wait.until(EC.url_changes("https://app.example.com/login"))

# 8. 标题变化
wait.until(EC.title_contains("仪表盘"))

# 9. 元素处于某状态（如 disabled）
wait.until(EC.element_to_be_enabled((By.ID, "submit")))
wait.until(EC.element_to_be_selected((By.CSS_SELECTOR, "#checkbox")))

显式等待的本质：「轮询直到条件满足或超时」。默认 0.5s 轮询一次，10s 内最多等 20 次。可以自定义轮询间隔：

wait = WebDriverWait(driver, 10, poll=1)   # 每秒轮询 1 次

隐式等待（全局兜底）

python — 隐式等待
driver.implicitly_wait(5)   # 全局：所有 find_element 最多等 5s

# 隐式等待 vs 显式等待
# 隐式：全局生效，所有 find_element 都等（简单但不够灵活）
# 显式：针对特定元素，等特定条件（灵活但代码多）

# 实际项目：两者混用
# 隐式 5s（兜底）+ 显式等待（关键操作）
driver.implicitly_wait(5)
wait = WebDriverWait(driver, 10)
el = wait.until(EC.element_to_be_clickable((By.ID, "btn")))   # 显式等"可点击"

隐式等待 + 显式等待冲突：如果设了 implicitly_wait(10)，invisibility_of_element_located 会等「10s × 10 = 100s」（因为元素不存在，隐式等待每次都等 10s 才报 Timeout）。建议：用显式等待就关掉隐式（implicitly_wait(0)）。

AJAX 等待（最常见的实际场景）

Web 应用大量用 AJAX 异步加载。「点按钮 → 等接口返回 → 等 DOM 更新」三步，Selenium 的默认等待（页面加载）不够。

python — AJAX 等待策略
# 场景：点"提交" → 等接口返回 → 等"提交成功"文字出现

# 策略 1：等"结果元素"出现（最常用）
driver.find_element(By.ID, "submit").click()
wait.until(EC.presence_of_element_located((By.ID, "success-msg")))
assert "提交成功" in driver.find_element(By.ID, "success-msg").text

# 策略 2：等"加载提示"消失（反例）
driver.find_element(By.ID, "submit").click()
wait.until(EC.invisibility_of_element_located((By.ID, "loading")))
# 加载提示消失 = 数据已渲染

# 策略 3：等 JS 变量（接口已返回）
wait.until(
    lambda d: d.execute_script("return window.appData") is not None
)

# 策略 4：等网络空闲（没有未完成的请求）
def wait_for_ajax(driver, timeout=30):
    """等所有 AJAX 请求完成"""
    start = time.time()
    while time.time() - start < timeout:
        ready_state = driver.execute_script("return document.readyState")
        ajax_count = driver.execute_script(
            "return jQuery.active"   # jQuery 在用的请求数
        )
        if ready_state == "complete" and ajax_count == 0:
            return True
        time.sleep(0.5)
    raise TimeoutError("AJAX 等待超时")

driver.find_element(By.ID, "submit").click()
wait_for_ajax(driver)
# 此时可以安全断言

动态内容等待（懒加载 / 虚拟滚动）

很多页面「滚动加载」（懒加载）或「虚拟滚动」（只渲染可见区域），元素不在 DOM 里 = 定位不到。需要「先滚动，再等」。

python — 懒加载等待
# 场景：列表有 100 条，滚动加载，找"第 80 条"

def scroll_to_element(driver, locator, timeout=10):
    """滚动到元素出现（懒加载）"""
    start = time.time()
    while time.time() - start < timeout:
        elements = driver.find_elements(*locator)
        if elements:
            driver.execute_script(
                "arguments[0].scrollIntoView({behavior: 'smooth', block: 'center'});",
                elements[0]
            )
            time.sleep(0.5)   # 等滚动完成
            elements = driver.find_elements(*locator)
            if elements:
                return elements[0]
        # 滚到底部（触发懒加载）
        driver.execute_script("window.scrollTo(0, document.body.scrollHeight)")
        time.sleep(0.5)
    raise TimeoutError("元素没出现（懒加载失败）")

# 用法
el = scroll_to_element(driver, (By.CSS_SELECTOR, ".item:nth-child(80)"))

自定义等待函数（实战最常用）

Selenium 内置的 expected_conditions 不够用时，自定义：

python — 自定义等待
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.common.by import By

def wait_for_condition(driver, condition, timeout=10, interval=0.5):
    """通用等待：等 condition 返回 truthy"""
    def _check(driver):
        return condition(driver) or False
    return WebDriverWait(driver, timeout, poll=interval).until(_check)

# 用法 1：等"某个元素的某个属性"
wait_for_condition(
    driver,
    lambda d: d.find_element(By.ID, "btn").get_attribute("disabled") is None,
    timeout=10
)

# 用法 2：等"列表加载完成"（不再变化）
def list_stable(driver):
    el = driver.find_element(By.ID, "list")
    current = el.get_attribute("data-count")
    return current  # 返回计数，wait_for_condition 等它不再变

# 实例：等"图表加载完成"（canvas 有内容）
wait_for_condition(
    driver,
    lambda d: d.execute_script(
        "return document.querySelector('#chart canvas').getContext('2d').getImageData(0,0,1,1).data[3] > 0;"
    )
)

等待策略「三不」原则：① 不用 time.sleep 等元素；② 不混用隐式等待 + 显式等待（用显式就关掉隐式）；③ 不「等」不存在的条件（如等「加载提示消失」，但页面根本没加载提示 → 超时）。
