用例数据放哪里

PC 自动化「用例数据」（输入、期望、路径）放三种地方：

数据源对比
# Excel（测试人员最熟，直接改）
# - 优点：不用写代码，直接改表格
# - 缺点：读要装 openpyxl，批量读慢
# 适合：测试数据"经常改"、非技术人员维护

# CSV（简单，纯文本）
# - 优点：轻量、纯文本、Git 友好
# - 缺点：列多时难读，没类型
# 适合：参数化数据、Git 管理

# JSON（结构化，嵌套）
# - 优点：支持嵌套（用例有子步骤）、类型明确
# - 缺点：写起来啰嗦
# 适合：复杂用例（多步骤、多条件）

数据驱动 + 参数化

把「用例数据」和「测试逻辑」分离，一条脚本跑 N 组数据：

python — CSV 数据驱动
# test_cases/excel_operations.csv
# operation,input_data,expected_result
新建文件,hello,文件内容=hello
修改单元格,A1=42,A1值=42
删除行,行3,总行数=99
保存,文件名=test,文件存在

import pytest
import csv

@pytest.fixture(scope="module")
def excel_cases():
    """读 Excel 操作用例（CSV）。"""
    cases = []
    with open("test_cases/excel_operations.csv", "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            cases.append(row)
    return cases

@pytest.mark.parametrize("case", excel_cases(), ids=lambda c: c["operation"])
def test_excel_operation(case, notepad_app):
    """Excel 操作（参数化驱动）。"""
    # 测试逻辑（一条脚本跑 N 组数据）
    # case = {"operation": "新建文件", "input_data": "hello", "expected_result": "文件内容=hello"}

    operation = case["operation"]
    input_data = case["input_data"]
    expected = case["expected_result"]

    print(f"执行：{operation}（输入：{input_data}）")

    # 执行操作
    result = execute_excel_operation(notepad_app, operation, input_data)

    # 断言
    assert result == expected, f"期望 {expected}，实际 {result}"

# 运行
# pytest test_excel.py -v
# 会跑 4 条（CSV 里 4 行）

import pytest
from openpyxl import load_workbook

@pytest.fixture(scope="module")
def excel_cases_xlsx():
    """读 Excel 用例数据。"""
    wb = load_workbook("test_cases/excel_test_data.xlsx")
    ws = wb.active

    cases = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[0] is None:
            continue
        cases.append({
            "case_id": row[0],
            "operation": row[1],
            "input": row[2],
            "expected": row[3],
        })
    return cases

@pytest.mark.parametrize(
    "case",
    excel_cases_xlsx(),
    ids=lambda c: f"{c['case_id']}_{c['operation']}"
)
def test_excel_from_xlsx(case, notepad_app):
    """Excel 数据驱动（用例从 Excel 读）。"""
    print(f"用例 {case['case_id']}：{case['operation']}")

    result = execute_operation(notepad_app, case["operation"], case["input"])
    assert result == case["expected"]

数据隔离：每个用例独立数据

python — 数据隔离
import pytest
import os
import uuid

@pytest.fixture
def isolated_data_dir(request):
    """每个用例独立数据目录（避免用例间互相影响）。"""
    # 唯一目录（用例名 + UUID）
    unique_id = f"{request.node.name}_{uuid.uuid4().hex[:8]}"
    data_dir = os.path.join("test_data", unique_id)
    os.makedirs(data_dir, exist_ok=True)

    yield data_dir

    # 清理（用例结束删数据）
    import shutil
    if os.path.exists(data_dir):
        shutil.rmtree(data_dir)

# 使用（每个用例独立数据）

conftest.py — 公共 fixture
import pytest
from openpyxl import load_workbook
import os
import uuid

@pytest.fixture(scope="session")
def excel_app():
    """启动 Excel（session 级，只启动一次）。"""
    import subprocess
    import time
    subprocess.Popen(["excel.exe"])
    time.sleep(5)
    yield
    import pyautogui
    pyautogui.hotkey("alt", "f4")
    time.sleep(1)

@pytest.fixture(scope="module")
def excel_cases():
    """读用例数据（Excel）。"""
    wb = load_workbook("test_data/excel_operations.xlsx")
    ws = wb.active
    cases = []
    for row in ws.iter_rows(min_row=2, values_only=True):
        if row[0] is None:
            continue
        cases.append({
            "case_id": row[0],
            "module": row[1],
            "operation": row[2],
            "input": row[3],
            "expected": row[4],
            "priority": row[5],
        })
    return cases

@pytest.fixture
def isolated_dir(request):
    """每个用例独立数据目录。"""
    unique_dir = os.path.join("test_data", f"{request.node.name}_{uuid.uuid4().hex[:8]}")
    os.makedirs(unique_dir, exist_ok=True)
    yield unique_dir
    import shutil
    shutil.rmtree(unique_dir, ignore_errors=True)
test_excel/test_file_ops.py — 文件操作用例
import pytest
from conftest import excel_cases, excel_app

@pytest.mark.excel
@pytest.mark.file_ops
@pytest.mark.parametrize("case", excel_cases, ids=lambda c: c["case_id"])
def test_file_operations(case, excel_app, isolated_dir):
    """文件操作（参数化驱动，按优先级筛）。"""
    # 只跑 file 模块
    if case["module"] != "file":
        pytest.skip("非文件操作用例")

    print(f"执行：{case['case_id']} - {case['operation']}")

    # 执行操作
    result = execute_operation(excel_app, case["operation"], case["input"], isolated_dir)

    # 断言
    assert result == case["expected"], \
        f"用例 {case['case_id']} 失败：期望 {case['expected']}，实际 {result}"

# 运行
# pytest test_excel/ -m "excel and file_ops" -v
# 只跑 Excel + 文件操作用例（C001、C002）

PC 自动化「用例管理」核心思路：数据驱动（用例数据放 Excel/CSV，参数化驱动）、分组（按功能模块 + 标记筛）、隔离（每个用例独立数据目录）、依赖（前置用例先跑）。一条脚本跑 N 组数据，改数据不用改代码，用例「可管理、可筛选、可追溯」。
