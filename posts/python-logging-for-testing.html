logging 基础：配置 + 记录

logging 是 Python 内置的日志模块。核心就两步：配置（basicConfig）→ 记录（logger.info/error）。

python — 基本用法
import logging

# 最简配置（一条搞定，默认 INFO 级别 + 控制台输出）
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s"
)

# 获取 logger（每个模块一个）
logger = logging.getLogger(__name__)

# 记录日志
logger.info("测试开始")
logger.warning("超时配置较低（5s）")
logger.error("接口 404：/api/orders/999999")
logger.critical("数据库连接失败，测试终止")
logger.debug("请求 URL：https://api.example.com/api/orders")   # 控制台不显示（INFO 级别）

日志级别：DEBUG（调试细节）< INFO（关键步骤）< WARNING（警告但能继续）< ERROR（错误但能继续）< CRITICAL（致命，必须停）。测试脚本一般用 INFO 级别（关键步骤都打），排查问题时切到 DEBUG。

五个级别：什么时候用什么

python — 日志级别场景
import logging

logging.basicConfig(level=logging.DEBUG, format="%(levelname)s: %(message)s")
logger = logging.getLogger(__name__)

# DEBUG：调试细节（开发/测试时看，CI 上关）
logger.debug("请求参数：username=admin, password=***")
logger.debug("接口耗时：156ms")

# INFO：关键步骤（"发生了什么"）
logger.info("开始测试：用户登录")
logger.info("登录成功，token=***...（前 20 位）")
logger.info("测试完成：5/5 通过")

# WARNING：警告但能继续（"有点异常，但没崩"）
logger.warning("超时配置低于推荐值（5s < 10s）")
logger.warning("第 3 次重试才成功（接口不稳定）")

# ERROR：错误但能继续（"这条失败了，但其他还能跑"）
logger.error("用例 test_payment 失败：金额溢出")
logger.error("接口 500：/api/orders/ORD-999")

# CRITICAL：致命（"整个测试终止"）
logger.critical("数据库连接失败，所有用例跳过")
logger.critical("配置文件缺失，测试无法进行")

格式：时间 + 级别 + 来源 + 消息

python — 日志格式
import logging

# 自定义格式（含文件名 + 行号）
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s:%(funcName)s:%(lineno)d - %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S"
)

logger = logging.getLogger("api_test")

logger.info("开始测试")
# 输出：
# 2026-11-23 10:00:00 [INFO] api_test:main:45 - 开始测试

日志写文件 + 控制台（两个 Handler）

python — 双 Handler
import logging
from logging.handlers import RotatingFileHandler
from datetime import datetime

# 创建 logger
logger = logging.getLogger("api_test")
logger.setLevel(logging.DEBUG)   # logger 级别（最低，所有都传过来）

# Handler 1：控制台（INFO 级别，看关键步骤）
console_handler = logging.StreamHandler()
console_handler.setLevel(logging.INFO)
console_format = logging.Formatter("[%(levelname)s] %(message)s")
console_handler.setFormatter(console_format)

# Handler 2：文件（DEBUG 级别，全部记录）
# 按大小轮转（10MB 一个文件，最多 5 个）
file_handler = RotatingFileHandler(
    "test_run.log",
    maxBytes=10 * 1024 * 1024,
    backupCount=5,
    encoding="utf-8"
)
file_handler.setLevel(logging.DEBUG)
file_format = logging.Formatter(
    "%(asctime)s [%(levelname)s] %(name)s:%(funcName)s:%(lineno)d - %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S"
)
file_handler.setFormatter(file_format)

# 添加 Handler
logger.addHandler(console_handler)
logger.addHandler(file_handler)

# 记录
logger.debug("请求 URL：https://api.example.com/api/orders")   # 只写文件
logger.info("测试开始")                                          # 文件 + 控制台
logger.error("用例失败")                                         # 文件 + 控制台

RotatingFileHandler 自动按大小轮转（10MB 一个文件），不会「无限增大」。测试跑一整天不会生成 10GB 日志。控制台用 INFO（看关键步骤），文件用 DEBUG（全记录，排查时翻）。

loguru：简化日志（可选）

from loguru import logger

# 一行配置（比 logging 简洁）
logger.add("test.log", level="DEBUG", rotation="10 MB", encoding="utf-8")

# 记录（不用 getLogger，直接用 logger）
logger.info("测试开始")

logger.info("用例执行中")

# 输出：
# 2026-11-23 10:00:00.123 | INFO     | test.py:45 - 测试开始
# 2026-11-23 10:00:00.456 | INFO     | test.py:48 - 用例执行中

loguru vs logging：logging 是标准库（不装也能用），loguru 是第三方（pip install 一下，API 更简洁）。测试脚本「快速打日志」用 loguru，「需要自定义 Handler」用 logging。

案例：接口测试全流程打日志

把上面学到的串起来，写一个「接口测试全流程打日志」的完整脚本：

python — 完整案例
import logging
import requests
from datetime import datetime
from logging.handlers import RotatingFileHandler

# 1. 配置日志
def setup_logger(name="api_test"):
    """配置 logger（控制台 + 文件双 Handler）"""
    logger = logging.getLogger(name)
    logger.setLevel(logging.DEBUG)

    # 控制台（INFO）
    console = logging.StreamHandler()
    console.setLevel(logging.INFO)
    console.setFormatter(logging.Formatter("[%(levelname)s] %(message)s"))
    logger.addHandler(console)

    # 文件（DEBUG）
    log_file = f"test_{datetime.now().strftime('%Y%m%d')}.log"
    file_handler = RotatingFileHandler(log_file, maxBytes=10*1024*1024, backupCount=3, encoding="utf-8")
    file_handler.setLevel(logging.DEBUG)
    file_handler.setFormatter(logging.Formatter(
        "%(asctime)s [%(levelname)s] %(message)s", datefmt="%Y-%m-%d %H:%M:%S"
    ))
    logger.addHandler(file_handler)

    return logger

logger = setup_logger()

# 2. 封装"带日志的"接口调用
def logged_request(method, url, **kwargs):
    """发请求 + 打日志"""
    logger.info(f"请求：{method} {url}")
    logger.debug(f"参数：{kwargs}")

    start = datetime.now()
    try:
        resp = requests.request(method, url, timeout=10, **kwargs)
        elapsed = (datetime.now() - start).total_seconds()
        logger.info(f"响应：{resp.status_code}（{elapsed*1000:.0f}ms）")
        return resp
    except requests.exceptions.Timeout:
        logger.error(f"请求超时：{url}")
        raise
    except requests.exceptions.ConnectionError as e:
        logger.error(f"连接失败：{url}（{e}）")
        raise

# 3. 测试流程
def test_user_flow():
    logger.info("=" * 50)
    logger.info("测试开始：用户流程")

    # 登录
    base_url = "https://api.example.com"
    resp = logged_request("POST", f"{base_url}/api/login",
        json={"username": "admin", "password": "123456"}
    )
    token = resp.json()["token"]
    logger.info(f"登录成功：token=***...")

    # 查订单
    resp = logged_request("GET", f"{base_url}/api/orders/ORD-001",
        headers={"Authorization": f"Bearer {token}"}
    )
    order = resp.json()
    logger.info(f"订单 {order['order_id']}：状态 {order['status']}，金额 {order['amount']}")

    # 支付
    resp = logged_request("POST", f"{base_url}/api/orders/{order['order_id']}/pay",
        json={"amount": order["amount"]},
        headers={"Authorization": f"Bearer {token}"}
    )
    pay_result = resp.json()
    if pay_result["status"] == "paid":
        logger.info("支付成功")
    else:
        logger.warning(f"支付异常：{pay_result.get('message')}")

    logger.info("测试完成")
    logger.info("=" * 50)

# 运行
if __name__ == "__main__":
    test_user_flow()

测试脚本「日志三原则」：① 关键步骤打 INFO（开始/结束/成功/失败）；② 调试细节打 DEBUG（请求参数、响应内容）；③ 错误打 ERROR（接口 4xx/5xx、超时、连接失败）。日志写文件（RotatingFileHandler 防无限增大），排查时翻文件，控制台只看 INFO。
