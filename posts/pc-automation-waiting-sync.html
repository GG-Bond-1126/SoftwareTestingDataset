四种同步策略对比

策略对比
# 策略 1：固定等待（time.sleep）
# - 最简单，一行代码
# - 慢（等多久是"猜的"）
# - 脆（界面快了等多 5s，慢了不够 5s）
# 适合：兜底、调试

# 策略 2：条件等待（轮询检查）
# - 等"某个条件成立"才继续
# - 快（条件成立了立即走）
# - 稳（不用猜等多久）
# 适合：等界面加载、等元素出现

# 策略 3：轮询 + 超时
# - 条件等待 + 最大等待时间
# - 条件不成立也"最多等 N 秒"
# - 最常用（90% 场景）
# 适合：等接口响应、等文件生成

# 策略 4：事件驱动（监听）
# - 注册"事件监听"（鼠标/键盘/窗口变化）
# - 事件触发时立即响应
# - 最复杂（要写监听器）
# 适合：监听用户操作、监听窗口状态

固定等待：time.sleep（简单但慢）

python — 固定等待
import pyautogui
import time

# 基本用法
pyautogui.press("enter")
time.sleep(2)   # 等 2 秒（"猜"界面需要 2 秒加载）

# pyautogui 的 PAUSE（每次操作后自动等）
pyautogui.PAUSE = 0.5
pyautogui.press("enter")   # 自动等 0.5s
pyautogui.press("enter")   # 自动等 0.5s

# 问题：
# - 界面 0.5s 加载完，你等了 2s（浪费 1.5s）
# - 界面 3s 才加载完，你只等 2s（"跑太快"失败）
# - 不同机器速度不同，"猜"的等待时间不准

条件等待：轮询检查（推荐）

python — 条件等待
import pyautogui
import time

def wait_for_condition(check_func, timeout=10, interval=0.5):
    """轮询等待条件成立。

    Args:
        check_func: 无参数函数，返回 bool（True = 条件成立）
        timeout: 最长等待时间（秒）
        interval: 检查间隔（秒）

    Returns:
        条件成立返回 True，超时返回 False
    """
    start = time.time()
    while time.time() - start < timeout:
        try:
            if check_func():
                print(f"  条件成立（{time.time() - start:.1f}s）")
                return True
        except Exception as e:
            print(f"  检查异常：{e}")
        time.sleep(interval)
    print(f"  超时（{timeout}s）")
    return False

# 场景 1：等界面元素出现（图像识别）
def file_saved():
    """检查"已保存"提示是否出现（截图找图）。"""

轮询 + 超时：最常用（90% 场景）

python — 轮询 + 超时
import pyautogui
import time

def wait_for_image(image_path, timeout=10, interval=0.5, confidence=0.9):
    """等图片出现（轮询 + 超时）。

    Args:
        image_path: 参考图路径
        timeout: 最长等待（秒）
        interval: 检查间隔（秒）
        confidence: 置信度

    Returns:
        图片位置 (x, y)，超时返回 None
    """
    start = time.time()
    while time.time() - start < timeout:
        location = pyautogui.locateOnScreen(image_path, confidence=confidence)
        if location:
            center = pyautogui.center(location)
            print(f"  找到图片（{time.time() - start:.1f}s）")
            return center
        time.sleep(interval)
    print(f"  图片 {image_path} 没出现（{timeout}s）")
    return None

# 使用：等"成功弹窗"出现
location = wait_for_image("success_popup.png", timeout=15, confidence=0.9)
if location:
    pyautogui.click(location)
    print("✅ 成功弹窗已点击")
else:
    print("❌ 成功弹窗没出现（15s 超时）")

# 批量：等 3 张图片按顺序出现
def wait_for_images_sequential(image_paths, timeout_each=10):
    """按顺序等图片出现。"""
    for path in image_paths:
        location = wait_for_image(path, timeout=timeout_each)
        if location is None:
            return False
        pyautogui.click(location)
    return True

# 使用：登录流程（3 步）
images = ["login_form.png", "password_input.png", "success_popup.png"]
success = wait_for_images_sequential(images, timeout_each=10)
if success:
    print("✅ 登录流程完成")
else:
    print("❌ 登录流程中断")

事件驱动：监听（复杂场景）

python — 事件监听（pynput）
# pip install pynput
from pynput import mouse, keyboard
import threading

# 监听鼠标点击（事件驱动）
click_count = 0
click_done = threading.Event()

def on_click(x, y, button, pressed):
    global click_count
    if pressed:
        click_count += 1
        print(f"第 {click_count} 次点击：({x}, {y})")
        if click_count == 3:
            click_done.set()   # 3 次点击后停止

listener = mouse.Listener(on_click=on_click)
listener.start()

# 触发操作（模拟 3 次点击）
import pyautogui
for _ in range(3):
    pyautogui.click(500, 300)
    time.sleep(0.5)

click_done.wait(timeout=5)   # 等事件
listener.stop()
print(f"共 {click_count} 次点击")

# 监听键盘（事件驱动）
key_pressed = threading.Event()

def on_key_press(key):
    global key_pressed
    if key == keyboard.Key.esc:
        key_pressed.set()   # ESC 键按下后停止

key_listener = keyboard.Listener(on_press=on_key_press)
key_listener.start()

# 等 ESC 键（超时 10s）
if not key_pressed.wait(timeout=10):
    print("10s 内没按 ESC，继续...")
key_listener.stop()

事件驱动最复杂（要写监听器 + 线程），只在「监听用户操作」或「监听窗口状态变化」时用。90% 的 PC 自动化场景用「条件等待 + 超时」就够了（轮询检查某条件，超时兜底）。

PC 自动化的「等待策略」选择

策略选择
# 场景 → 推荐策略
# - 点击按钮后等弹窗 → 条件等待（等弹窗出现）
# - 等待文件生成 → 轮询 + 超时（等文件存在）
# - 等待窗口标题变化 → 条件等待（检查标题）
# - 等待网络响应 → 固定等待（1~2s 兜底）
# - 监听用户操作 → 事件驱动（pynput）
# - 兜底（不确定等多久）→ 固定等待（time.sleep）

案例：等待界面加载完成的完整流程

python — 完整案例
import pyautogui
import uiautomation as auto
import time
import os

# 1. 启动应用（固定等待 1s 启动）
import subprocess
subprocess.Popen(["notepad.exe"])
time.sleep(1)

# 2. 等窗口就绪（条件等待）
def window_ready():
    notepad = auto.WindowControl(searchDepth=1, subTitle="记事本")
    if not notepad.Exists(1):
        return False
    # 窗口就绪：不含"加载中"
    return "加载中" not in notepad.Name

notepad = wait_for_condition(window_ready, timeout=15)
if not notepad:
    print("❌ 记事本窗口没就绪（15s）")
    exit(1)

# 3. 输入文本 + 保存
edit_box = notepad.child_window(control_type="Edit")
edit_box.set_edit_text("测试等待 + 同步")

# 4. 保存（Ctrl+S）
pyautogui.hotkey("ctrl", "s")

# 5. 等"已保存"提示（条件等待 + 图像识别）
def saved_toast_visible():
    location = pyautogui.locateOnScreen("saved_toast.png", confidence=0.9)
    return location is not None

if not wait_for_condition(saved_toast_visible, timeout=10):
    print("⚠️ 保存提示没出现（可能没弹窗）")
else:
    print("✅ 保存成功（提示出现）")

# 6. 等文件存在（轮询 + 超时）
save_path = os.path.join(os.path.expanduser("~"), "Desktop", "untitled.txt")

def file_exists():
    return os.path.exists(save_path)

if not wait_for_condition(file_exists, timeout=15, interval=1.0):
    print(f"❌ 文件 {save_path} 没生成（15s）")
else:
    print(f"✅ 文件已生成：{save_path}")

# 7. 关闭
pyautogui.hotkey("alt", "f4")

print("\n=== 等待与同步测试完成 ===")

PC 自动化「等待三原则」：① 固定等待只做兜底（启动应用、等 1 秒），核心操作不用 sleep；② 条件等待是主力（轮询检查某条件，成立立即走）；③ 加超时（条件不成立也「最多等 N 秒」，避免死循环）。100 条用例跑下来，「固定等待」省 50% 时间，「条件等待 + 超时」稳 90%。
