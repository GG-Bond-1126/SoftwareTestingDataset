import 的三种基本方式

Python 的「模块」就是「一个 .py 文件」。import 就是「把别人写好的代码拉进来用」。

python — import 基本方式
# 假设有个文件：test_utils.py
# 内容：
# def login(username, password):
#     ...
# def create_order(items):
#     ...

# 方式一：import 整个模块（最常用）
import test_utils
test_utils.login("admin", "123456")
test_utils.create_order([{"name": "手机壳", "qty": 2}])

# 方式二：from 导入指定函数（更简洁）
from test_utils import login, create_order
login("admin", "123456")
create_order([{"name": "手机壳", "qty": 2}])

# 方式三：from 导入所有（不推荐，命名空间污染）
from test_utils import *
login("admin", "123456")   # 不用写模块名前缀

# 方式四：改别名（常用库太长的名字）
import pandas as pd
import matplotlib.pyplot as plt
import requests as rq

99% 的场景用方式一（import 整个模块）。方式二（from import）适合「只导入一两个函数」且名字不冲突时。方式三（import *）基本不用，命名空间太乱。

__name__ == 「__main__」：让脚本「既能被 import，也能直接运行」

这是测试脚本的「入口」判断。一个 .py 文件既可以是「被 import 的模块」，也可以是「直接运行的脚本」：

python — __name__ 判断
# test_runner.py（既能被 import，也能直接运行）

def run_test_suite(tests):
    """跑一组测试（可以被其他模块 import）"""
    results = []
    for test in tests:
        passed = test()
        results.append(passed)
    return results

def main():
    """直接运行时的入口"""
    print("=== 测试套件 ===")
    results = run_test_suite([
        test_login,
        test_order,
        test_payment,
    ])
    passed = sum(results)
    print(f"总计 {len(results)} 条，通过 {passed} 条")

# 关键判断
if __name__ == "__main__":
    # 直接运行 python test_runner.py 时，__name__ 是 "__main__"
    main()
    # 被 import 时，__name__ 是模块名（如 "test_runner"），不执行 main()

测试脚本的标准写法：函数定义 + if __name__ == "__main__": main()。这样别人 import 你的模块时，不会自动执行 main（只拿到函数），直接运行时才执行。

包（package）和 __init__.py

项目大了，「一个文件装不下」，就把文件组织成「文件夹 + __init__.py」，这叫「包」。

项目结构
test_project/
├── __init__.py          # 标记"这是包"（可以空文件）
├── api_client.py        # 接口客户端
├── data_factory.py      # 数据工厂
├── utils.py             # 工具函数
└── tests/
    ├── __init__.py      # 测试包
    ├── test_login.py
    └── test_order.py
python — 包内 import
# 在包外导入（用"点"表示层级）
from test_project.api_client import ApiClient
from test_project.utils import load_config

# 包内导入（相对导入，用点表示"当前包"）
# test_project/utils.py 里
from . import api_client       # 导入同包的 api_client
from .api_client import ApiClient  # 导入同包的某个类

__init__.py 是「包的身份证」。没有它，Python 不认为这个文件夹是包（不能 import）。可以写空文件，也可以写「导出逻辑」（控制 from 包 import * 时导出哪些名字）。

循环导入问题

python — 循环导入
# a.py
from b import foo   # a 要 b 里的 foo

# b.py
from a import bar   # b 要 a 里的 bar
# → 循环导入！a 导入 b 时，b 又导入 a，a 还没加载完，报错

# 解法一：延迟导入（函数内部 import）
# a.py
def use_b():
    from b import foo   # 函数运行时才导入，不在模块加载时
    return foo()

# 解法二：把公共部分抽到第三个模块
# c.py（公共定义）
def foo(): ...
def bar(): ...

# a.py
from c import foo, bar   # a 和 b 都从 c 导入，不互相依赖
# b.py
from c import foo, bar

案例：把登录方法抽成公共模块

测试项目里「登录方法」被 10 条用例引用。把登录逻辑抽成公共模块，所有用例共用：

项目结构
test_project/
├── __init__.py
├── config.py          # 配置（base_url、超时）
├── api_client.py      # 接口客户端（封装 requests）
├── data_factory.py    # 数据工厂（造测试用户）
└── tests/
    ├── __init__.py
    ├── conftest.py    # pytest fixture
    ├── test_login.py
    ├── test_order.py
    └── test_payment.py
config.py — 配置模块
import json
import os

def load_config():
    """读配置文件，没有就用默认值"""
    config_path = os.path.join(os.path.dirname(__file__), "config.json")
    try:
        with open(config_path, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        return {
            "base_url": "https://api.example.com",
            "timeout": 10,
            "retries": 3,
        }
api_client.py — 接口客户端
import requests
from config import load_config

class ApiClient:
    """通用接口客户端（所有测试共用）"""

    def __init__(self, config=None):
        self.config = config or load_config()
        self.base_url = self.config["base_url"]
        self.timeout = self.config.get("timeout", 10)
        self.session = requests.Session()

    def login(self, username, password):
        """登录（10 条用例都调这个方法）"""
        resp = self.session.post(
            f"{self.base_url}/api/login",
            json={"username": username, "password": password},
            timeout=self.timeout
        )
        resp.raise_for_status()
        token = resp.json().get("token")
        self.session.headers["Authorization"] = f"Bearer {token}"
        return token

    def get(self, endpoint, **kwargs):
        resp = self.session.get(
            f"{self.base_url}{endpoint}",
            timeout=self.timeout,
            **kwargs
        )
        resp.raise_for_status()
        return resp.json()

    def post(self, endpoint, data=None, **kwargs):
        resp = self.session.post(
            f"{self.base_url}{endpoint}",
            json=data,
            timeout=self.timeout,
            **kwargs
        )
        resp.raise_for_status()
        return resp.json()
tests/test_order.py — 用公共模块
import pytest
from api_client import ApiClient

# pytest fixture：每个测试前创建客户端
@pytest.fixture
def client():
    return ApiClient()

def test_create_order(client):
    """下单（登录逻辑复用 api_client 里的 login）"""
    token = client.login("admin", "123456")
    order = client.post("/api/orders", data={
        "user_id": 1001,
        "items": [{"name": "手机壳", "qty": 2, "price": 49.99}]
    })
    assert order.get("order_id")

def test_payment(client):
    """支付（同样复用 login）"""
    token = client.login("admin", "123456")
    result = client.post("/api/orders/ORD-001/pay", data={"amount": 99.99})
    assert result.get("status") == "paid"

测试项目「不重复写」的三层结构：config.py（配置集中管）、api_client.py（接口方法封装一次，所有用例共用）、tests/（每条用例只写「业务逻辑」，不用重复写「怎么发请求」）。登录方法写了 10 遍 vs 1 遍，维护成本差 10 倍。
