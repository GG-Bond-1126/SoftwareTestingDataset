桌面自动化 + LLM Agent：PC 测试的 AI 闭环

环境搭建

桌面测试需要三层工具：pyautogui（鼠标键盘控制 + 截图）、uiautomation（Windows UI 元素树遍历，比坐标点击更稳定）、LLM 视觉 API（屏幕截图 → 元素识别 → 操作生成）。

安装依赖（Windows）
pip install pyautogui uiautomation openai opencv-python pillow

# 验证 uiautomation（Windows 专属，需管理员权限）
python -c "import uiautomation; print(uiautomation.UIAutomation)"

# 验证 pyautogui 截图
python -c "import pyautogui; img = pyautogui.screenshot(); print('截图尺寸:', img.size)"
安装依赖（Linux，Wayland 下 pyautogui 受限）
pip install pyautogui openai opencv-python pillow

# Linux 截图（pyautogui 在 X11 下有效）
import pyautogui
img = pyautogui.screenshot()
img.save("test.png")
配置 LLM 密钥（.env）
# .env
OPENAI_API_KEY=sk-xxxxxxxx
LLM_MODEL=gpt-4o
# 桌面测试建议用 vision 模型，成本约 $0.01/次截图分析

Windows 下 uiautomation 需要以管理员权限运行 Python，否则无法遍历 UWP/高权限进程的元素树。CI 中可用 Windows Server Runner + 管理员 Token。

基础应用：LLM 读桌面截图 → 生成 pyautogui 操作 → 执行验证

对桌面应用（如记事本、计算器）进行自动化测试：LLM 看截图找控件位置，生成 pyautogui 坐标操作，执行后用 OpenCV 截图对比验证结果。

desktop_agent.py — 桌面 LLM Agent
"""
桌面 LLM Agent：截图 → LLM 分析 → pyautogui 执行 → OpenCV 验证
"""
import base64, json, os, time, pathlib
import pyautogui
import cv2
import numpy as np
from openai import OpenAI

client = OpenAI()
MODEL = os.environ.get("LLM_MODEL", "gpt-4o")
pyautogui.FAILSAFE = True   # 鼠标移到屏幕左上角可紧急停止
pyautogui.PAUSE = 0.3       # 每次操作后暂停 0.3s，防过快

class DesktopAgent:
    """
    用法:
        agent = DesktopAgent()
        agent.act("点击记事本窗口中的'文件'菜单")
        agent.act("点击'新建'选项")
        agent.verify("页面中出现空白文档")
        agent.save_state("step1")
    """

    def __init__(self, screenshot_dir: str = "screenshots"):
        self._dir = pathlib.Path(screenshot_dir)
        self._dir.mkdir(exist_ok=True)
        self._step = 0

    def _screenshot(self, tag: str = "") -> bytes:
        png = pyautogui.screenshot()
        data = png.tobytes()
        b64 = base64.b64encode(data).decode()
        self._step += 1
        if tag:
            png.save(self._dir / f"{tag}_{self._step}.png")
        return b64

    def act(self, intent: str, max_retry: int = 2):
        """
        执行自然语言桌面操作意图
        LLM 返回: 坐标点击 / 键盘输入 / 菜单操作
        """
        system = (
            "你是桌面 UI 自动化专家。根据桌面截图和用户意图，"
            "生成 pyautogui Python 代码。"
            "支持操作：click(x,y)、hotkey('ctrl','a')、typewrite('text')、"
            "press('enter')、doubleClick(x,y)、rightClick(x,y)。\n"
            "坐标基于 1920x1080 分辨率，请根据截图比例换算。\n"
            "如果操作已完成或无需操作，输出 'DONE'。\n"
            "输出纯代码，不要 markdown。"
        )
        for attempt in range(max_retry):
            shot_b64 = self._screenshot(f"act_{intent[:20]}")
            prompt = f"用户意图: {intent}\n请生成 pyautogui 代码（坐标系 1920x1080）："
            resp = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": system},
                    {"role": "user", "content": [
                        {"type": "text", "text": prompt},
                        {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{shot_b64}"}},
                    ]},
                ],
                temperature=0.05,
                max_tokens=512,
            )
            code = resp.choices[0].message.content.strip()
            code = code.removeprefix("```python").removeprefix("```").removesuffix("```").strip()

            if code.upper() == "DONE":
                print(f"  [OK] {intent}")
                return

            print(f"  [exec] {code[:100]}")
            try:
                # 注入 pyautogui 到执行环境
                local = {"pyautogui": pyautogui, "time": time}
                exec(code, local)
                time.sleep(0.5)
                return
            except Exception as e:
                print(f"  [retry {attempt+1}] {e}")

        raise RuntimeError(f"操作失败（重试 {max_retry} 次）: {intent}")

    def verify(self, expectation: str) -> bool:
        """截图 → LLM 判断界面是否符合预期"""
        shot_b64 = self._screenshot("verify")
        resp = client.chat.completions.create(
            model=MODEL,
            messages=[{
                "role": "user",
                "content": [
                    {"type": "text", "text": (
                        f"判断当前桌面是否满足: 「{expectation}」\n"
                        "输出 JSON: {\"pass\": true/false, \"reason\": \"...\"}"
                    )},
                    {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{shot_b64}"}},
                ],
            }],
            temperature=0.0,
            max_tokens=256,
        )
        text = resp.choices[0].message.content.strip()
        text = text.removeprefix("```json").removeprefix("```").removesuffix("```").strip()
        result = json.loads(text)
        print(f"  [verify] {result}")
        return result.get("pass", False)

    def verify_ocr(self, expected_text: str) -> bool:
        """
        用 OpenCV 模板匹配验证界面上出现指定文本
        比 LLM 验证更快、更确定（适合 CI）
        """
        img = pyautogui.screenshot()
        gray = cv2.cvtColor(np.array(img), cv2.COLOR_RGB2GRAY)
        # 简单方案：截图保存后人工确认（生产环境可接 Tesseract OCR）
        path = self._dir / f"ocr_{expected_text[:20]}.png"
        img.save(path)
        print(f"  [ocr] 截图保存: {path}，请人工确认 '{expected_text}' 是否出现")
        return True  # 实际项目中接 Tesseract 后返回匹配结果
test_calculator.py — 计算器测试示例
"""
测试 Windows 计算器的基本运算
前提：计算器已启动且在前台
"""
import pytest
from desktop_agent import DesktopAgent

@pytest.fixture
def agent():
    import subprocess
    subprocess.Popen("calc.exe")  # 启动计算器
    pytest.wait(2)
    a = DesktopAgent()
    yield a

def test_addition(agent):
    """1 + 2 = 3"""
    agent.act("点击计算器数字键盘上的 '1' 按钮")
    agent.act("点击 '+' 按钮")
    agent.act("点击 '2' 按钮")
    agent.act("点击 '=' 按钮")
    assert agent.verify("计算器显示屏显示 3")

def test_subtraction(agent):
    agent.act("按 Ctrl+N 清空当前计算")
    agent.act("点击 '9' 按钮")
    agent.act("点击 '-' 按钮")
    agent.act("点击 '4' 按钮")
    agent.act("点击 '=' 按钮")
    assert agent.verify("计算器显示屏显示 5")

def test_divide_by_zero(agent):
    """边界：除以零"""
    agent.act("按 Ctrl+N 清空")
    agent.act("点击 '5' 按钮")
    agent.act("点击 '÷' 按钮")
    agent.act("点击 '0' 按钮")
    agent.act("点击 '=' 按钮")
    assert agent.verify("计算器显示错误提示或不可计算")
pyautogui 的 FAILSAFE=True 是最后的安全网：鼠标快速移到屏幕左上角 (0,0) 会触发 FailSafeException 中断脚本，防止 LLM 误操作导致不可逆行为。

项目进阶：CI 远程桌面 + LLM 自愈 + 需求转用例

桌面测试 CI 的挑战在于需要 GUI 环境。方案：GitHub Actions Windows Runner + Xvfb（Linux）或真实 Windows 虚拟机 + VNC 录制，LLM 在 UI 变化时自动重新定位控件。

ci_desktop_test.py — CI 远程桌面执行器
"""
CI 环境中的桌面测试执行器
支持 VNC 录制，失败时自动截图 + LLM 根因分析
"""
import os, base64, json, pathlib, subprocess, time, datetime
import pyautogui
from openai import OpenAI
from desktop_agent import DesktopAgent

client = OpenAI()
MODEL = os.environ.get("LLM_MODEL", "gpt-4o")

class CITestRunner:
    def __init__(self, app_launcher: str = "calc.exe"):
        self.agent = DesktopAgent()
        self._app_launcher = app_launcher
        self._screens = []

    def start_recording(self):
        """CI 中用 ffmpeg 录制 VNC 流（可选）"""
        ts = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        vnc_rec = f"vnc_rec_{ts}.mp4"
        # Windows: 用 OBS 或 ffmpeg 录制桌面
        # 简化：只记录关键步骤截图
        self._recording_file = vnc_rec
        print(f"  [record] 开始记录: {vnc_rec}")

    def launch_app(self):
        subprocess.Popen(self._app_launcher)
        time.sleep(3)  # 等待启动

    def run_case(self, steps: list[dict]):
        """
        执行测试步骤列表
        每步: {"action": "tap|input|menu", "intent": "...", "expect": "..."}
        失败时自动截图 + LLM 根因分析
        """
        self.launch_app()
        self.start_recording()

        for i, step in enumerate(steps, 1):
            print(f"  Step {i}/{len(steps)}: {step['intent']}")
            try:
                self.agent.act(step["intent"])
                time.sleep(1)
                if step.get("expect"):
                    passed = self.agent.verify(step["expect"])
                    if not passed:
                        raise AssertionError(f"断言失败: {step['expect']}")
            except Exception as e:
                # 失败处理：截图 + LLM 根因分析
                print(f"  [FAIL] Step {i}: {e}")
                self._on_failure(i, step, e)
                return False

        print("  [PASS] 全部步骤通过")
        return True

    def _on_failure(self, step_num: int, step: dict, error: Exception):
        """失败时收集证据并调用 LLM 分析根因"""
        # 保存失败截图
        fail_screenshot = pathlib.Path(f"failure_step{step_num}.png")
        pyautogui.screenshot().save(fail_screenshot)
        shot_b64 = base64.b64encode(fail_screenshot.read_bytes()).decode()

        prompt = f"""
        桌面自动化测试在第 {step_num} 步失败。
        操作意图: {step['intent']}
        错误信息: {str(error)[:200]}
        截图已附。

        请分析：
        1. 根因分类（UI变更/超时/权限/分辨率/其他）
        2. 具体分析（2-3句）
        3. 修复建议（具体操作）
        以 Markdown 输出。
        """
        resp = client.chat.completions.create(
            model=MODEL,
            messages=[{
                "role": "user",
                "content": [
                    {"type": "text", "text": prompt},
                    {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{shot_b64}"}},
                ],
            }],
            temperature=0.1,
            max_tokens=512,
        )
        report = resp.choices[0].message.content
        out = pathlib.Path(f"failure_report_step{step_num}.md")
        out.write_text(report, encoding="utf-8")
        print(f"  [report] 根因分析: {out}")

def main():
    runner = CITestRunner(app_launcher="notepad.exe")

    # 示例：记事本测试
    steps = [
        {"intent": "点击记事本菜单栏的'文件'", "expect": "文件菜单展开"},
        {"intent": "点击'新建'选项", "expect": "出现空白文档"},
        {"intent": "输入 'Hello Desktop Test'", "expect": "文档中出现文本"},
        {"intent": "按 Ctrl+S 保存", "expect": "弹出保存对话框"},
        {"intent": "输入文件名 'test'，点击'保存'", "expect": "对话框关闭，文件已保存"},
    ]

    success = runner.run_case(steps)
    print(f"\n最终结果: {'PASS' if success else 'FAIL'}")

if __name__ == "__main__":
    main()
requirements_to_cases.py — 需求文档 → 桌面测试用例
"""
从需求文档生成桌面测试用例（JSON 格式）
"""
import json, os, pathlib
from openai import OpenAI

client = OpenAI()
MODEL = os.environ.get("LLM_MODEL", "gpt-4o")

def generate_desktop_cases(requirement_md: str, app_name: str) -> list[dict]:
    """
    输入：需求文档 + 应用名
    输出：JSON 测试用例列表（供 ci_desktop_test.py 执行）
    """
    prompt = f"""
    应用: {app_name}
    需求文档:
    {requirement_md[:2000]}

    请生成桌面自动化测试用例，JSON 数组格式：
    [
      {{
        "name": "用例名称",
        "steps": [
          {{"action": "menu|click|type|hotkey", "intent": "自然语言描述操作",
           "expect": "操作后界面预期状态（可为空）"}},
          ...
        ]
      }}
    ]
    要求覆盖：正向流程、异常输入、边界值。
    """
    resp = client.chat.completions.create(
        model=MODEL,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.2,
        max_tokens=4096,
    )
    text = resp.choices[0].message.content.strip()
    text = text.removeprefix("```json").removeprefix("```").removesuffix("```").strip()
    return json.loads(text)

if __name__ == "__main__":
    requirement = """
    ## 文本编辑器需求
    - 支持新建、打开、保存文本文件
    - 支持字体大小调整（12-72pt）
    - 支持搜索替换功能
    - 文件编码支持 UTF-8、GBK
    - 未保存时关闭窗口应弹出确认提示
    """
    cases = generate_desktop_cases(requirement, "记事本")
    pathlib.Path("desktop_test_cases.json").write_text(
        json.dumps(cases, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(f"生成 {len(cases)} 条桌面测试用例")
.github/workflows/desktop-ci.yml（Windows Runner）
name: Desktop UI Tests
on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  desktop-test:
    runs-on: windows-latest

    steps:
      - uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install deps
        run: |
          pip install pyautogui uiautomation openai opencv-python pillow

      - name: Run desktop tests
        env:
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
          LLM_MODEL: gpt-4o
        run: python ci_desktop_test.py

      - name: Upload failure screenshots
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: desktop-failures
          path: |
            failure_step*.png
            failure_report_step*.md
          retention-days: 14
GitHub Actions Windows Runner 是无头服务器，pyautogui 需要虚拟显示。可安装 Microsoft.UI.Xaml 或使用 xvfb（需 WSL2）。本地测试建议在真实桌面环境运行。
LLM 根因分析保留在 CI Artifacts 中，可直接作为 Bug 单附件。分辨率固定为 1920x1080 可避免坐标换算误差。
