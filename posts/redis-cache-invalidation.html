import mysql.connector, threading

def deduct(uid, qty):

缓存三大问题，先分清是哪一种

问题一句话描述最直接的验证动作
缓存穿透查一个根本不存在的 key，缓存永远没有，每次请求都穿透到数据库反复查一个不存在的 id，看数据库是否每次都真被访问
缓存击穿一个热 key 恰好过期的那一瞬间，大量并发同时打到数据库让某个热门 key 过期，瞬间发一波并发请求，观察数据库连接数/响应时间
缓存雪崩大批 key 同时过期（或缓存服务整体挂掉），流量瞬间全压到数据库让一批 key 在同一时刻集中过期，或直接把缓存实例停掉，观察数据库压力曲线

三者的区别就在「量」和「性质」：穿透是查不存在的，击穿是单个热 key 过期，雪崩是大批 key 一起过期或缓存整体不可用。测试脚本要分别对应着设计，不能混着跑。

每种问题的测试方法 + 验证脚本

穿透：反复查一个不存在的 key

test_cache_penetration.py
import redis

r = redis.Redis(host="127.0.0.1", port=6379, db=0)
GHOST_KEY = "user:999999999"   # 保证数据库里也不存在的 id

def main():
    r.delete(GHOST_KEY)
    hits = misses = 0
    for _ in range(100):
        if r.get(GHOST_KEY):
            hits += 1
        else:
            misses += 1
    print(f"100 次查询：命中 {hits} 次，未命中 {misses} 次")
    # 如果业务做了"负缓存"（查不到就缓存一个空值/TTL 兜底），
    # 第 2 次起应该全部"命中"（命中的是那个空值），misses 最多只有 1
    # 如果 misses 一直是 100，说明穿透没有任何兜底，数据库每次都在白白被访问
if __name__ == "__main__":
    main()

击穿：热 key 过期瞬间的并发冲击

test_cache_breakdown.py
import threading
import redis

r = redis.Redis(host="127.0.0.1", port=6379, db=0)
HOT_KEY = "product:1"

def main():
    r.set(HOT_KEY, "v1", ex=2)   # 热 key，2 秒后过期
    import time
    time.sleep(2)                # 等到 key 真的没了
    # 此时模拟 50 个并发同时请求这个热 key
    results = []
    def worker():
        results.append(r.get(HOT_KEY))
    threads = [threading.Thread(target=worker) for _ in range(50)]
    for t in threads: t.start()
    for t in threads: t.join()
    none_count = sum(1 for v in results if v is None)
    print(f"key 过期后的并发请求里，有 {none_count} 个直接打到数据库（未命中）")
    # 如果业务用了"互斥锁重建缓存"或"逻辑过期"策略，none_count 应该远小于 50，
    # 只有 1 个请求真正去查库并回填，其余等待结果

雪崩：大批 key 同一时刻过期 / 缓存整体不可用

test_cache_avalanche.py
import redis, time, threading

r = redis.Redis(host="127.0.0.1", port=6379, db=0)

def main():
    # 1. 让 1000 个 key 在同一时刻集中过期（同一 TTL，同一秒写入）
    for i in range(1000):
        r.set(f"item:{i}", "v", ex=3)
    time.sleep(3)   # 全部过期
    alive = 0
    for i in range(1000):
        if r.get(f"item:{i}"):
            alive += 1
    print(f"集中过期后还活着的 key 数 = {alive}（预期 0，说明"雪崩式同时失效"确实发生了）")
    # 2. 更硬核的验证：直接停掉整个缓存实例，观察业务侧
    #    是否出现大量请求直接打到数据库、数据库连接池是否被打满、接口响应时间是否飙升
if __name__ == "__main__":
    main()

三种缓存更新策略，各该测什么

光测「三大问题」还不够，业务到底用哪种「更新/失效」策略，决定了要重点验证的边界不一样：

策略做法测试重点
Cache Aside（旁路缓存，最常用）读：先查缓存，没有再查库并回填；写：先更新数据库，再删除缓存写库成功但删缓存失败（网络抖动/缓存挂了）时，会不会长期留着旧值；并发读写交错下有没有「读到过期但还没被删的旧值」的窗口期
Read Through读请求由缓存组件自己负责「查不到就去查库并回填」，业务代码不用手写这套验证「回填」这一步在缓存侧是否真的原子完成（两个并发读同一个不存在的 key，是否都去查了一次库、还是只有一个查了另一个直接等结果）
Write Through写请求同时写缓存和数据库，保证「两个都成功」才返回成功验证「一致性」：写操作后立刻读，缓存和数据库必须一致；任何一方写失败，整个写操作要返回失败，不能出现「缓存更新了、数据库没更新」这种半成功状态
test_cache_aside.py — 模拟 Cache Aside 写后立刻读
import redis

r = redis.Redis(host="127.0.0.1", port=6379, db=0)
KEY = "user:1001:name"

def main():
    r.set(KEY, "alice", ex=3600)

    # 模拟"写库成功、删缓存失败"的异常路径：
    # 手动让"删除"这一步失败（比如操作一个不存在的 key 模拟网络抖动），
    # 看缓存里的旧值会残留多久
    r.set(KEY, "alice-updated")
    try:
        r.delete("user:1001:name")   # 这里故意删错 key 名（拼写错误），模拟"删除失败"
    except Exception:
        pass
    print(f"删除失败后，缓存里的值 = {r.get(KEY)}")
    # 期望：能发现这个"旧值残留"会一直存在到 TTL 到期，验证业务是否接受这种最长不一致窗口期
if __name__ == "__main__":
    main()

生产环境监控该盯什么

  • 缓存命中率：突然大幅下跌，往往意味着穿透/击穿正在发生，或者大批 key 集中过期（雪崩前兆）；
  • 缓存服务的内存使用率：接近上限且有大量 key 即将集中过期，要提前排查 TTL 是否设置得太整齐（都用了同一个固定值导致「同一秒集体失效」）；
  • 关键业务的「写后立刻读」路径，建议单独打点，跟踪「读到的旧值」出现的频率和持续时间，而不只是看缓存命中率这一个笼统的数字。
