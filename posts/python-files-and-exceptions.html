文件读写：open 和 with

测试脚本最常做的文件操作：读配置文件、写测试日志、导数据。Python 的文件操作核心就一个函数 open，但用法有讲究。

基本读写

python — 文件读写
# 读文件（"r" = read）
with open("config.txt", "r", encoding="utf-8") as f:
    content = f.read()        # 一次读完（文件不大时）
    # 或者逐行读（文件很大时）
    # lines = f.readlines()   # 读成列表
    # for line in f:          # 逐行迭代（内存最省）
    #     print(line.strip())

# 写文件（"w" = write，会覆盖原内容）
with open("result.txt", "w", encoding="utf-8") as f:
    f.write("测试通过\n")
    f.write("测试失败\n")

# 追加（"a" = append，不覆盖）
with open("result.txt", "a", encoding="utf-8") as f:
    f.write("新增一条日志\n")

with 语句（最推荐）：自动关闭文件

python — with vs 手动关闭
# 手动关闭（不推荐，容易忘）
f = open("data.txt", "r")
content = f.read()
f.close()   # 必须手动关闭，忘写就泄漏文件句柄

# 异常时也不会关闭（崩溃就泄漏）
# f = open("data.txt", "r")
# content = f.read()   # 这里报错，f.close() 没执行！

# with 自动关闭（推荐）
# 不管有没有异常，退出 with 块时自动关闭
with open("data.txt", "r") as f:
    content = f.read()
# 这里 f 已经自动关闭了

99% 的场景用 with 语句（with open(...) as f）。它保证文件自动关闭，不管有没有异常。手动 open + close 只在你需要「打开后不立即读」的场景用。

文件编码（中文必备）

python — 编码处理
# 读中文文件（必须指定 encoding="utf-8"，Windows 默认 GBK）
with open("用户数据.txt", "r", encoding="utf-8") as f:
    for line in f:

# 写文件也指定编码
with open("测试报告.txt", "w", encoding="utf-8") as f:

# Windows 上不加 encoding 可能报 UnicodeDecodeError

异常处理：try / except / finally

接口超时、文件不存在、JSON 解析失败——这些都是「异常」。不处理就脚本崩了，处理了就能「优雅失败」。

基本 try/except

python — try/except
import json

# 捕获特定异常（推荐）
try:
    with open("config.json", "r", encoding="utf-8") as f:
        data = json.load(f)
except FileNotFoundError:
    print("配置文件不存在，使用默认配置")
    data = {"timeout": 10}
except json.JSONDecodeError as e:
    print(f"配置 JSON 格式错误：{e}")
    data = {"timeout": 10}

# 捕获所有异常（最后手段，一般不用）
try:
    result = 1 / 0
except Exception as e:
    print(f"未知错误：{e}")

else 和 finally

python — else / finally
# else：没异常才执行
try:
    data = json.load(open("config.json"))
except (FileNotFoundError, json.JSONDecodeError):
    data = {}
else:
    # 解析成功才执行
    print(f"超时配置：{data.get('timeout')}")

# finally：无论有没有异常都执行（清理资源）
try:
    f = open("report.txt", "w")
    f.write("测试通过\n")
except Exception as e:
    print(f"写报告失败：{e}")
finally:
    # 无论成功失败都关闭文件
    try:
        f.close()
    except NameError:
        pass   # f 没创建成功也会走这里
    print("清理完成")

raise 自定义异常

python — 自定义异常
# 定义测试专用异常
class TestConfigError(Exception):
    """测试配置错误"""
    pass

class TestDataError(Exception):
    """测试数据错误"""
    pass

def load_config(config_path):
    """加载配置，出错抛自定义异常"""
    try:
        with open(config_path, "r", encoding="utf-8") as f:

    except FileNotFoundError:

try:

测试工程师的五个常用异常

python — 常用异常速查
# 1. FileNotFoundError：文件不存在
try:
    open("nonexistent.txt", "r")
except FileNotFoundError:
    print("文件不存在")

# 2. KeyError：字典键不存在
data = {"name": "test"}
try:
    print(data["email"])
except KeyError as e:
    print(f"缺少字段：{e}")

# 3. IndexError：列表索引越界
items = [1, 2, 3]
try:
    print(items[5])
except IndexError:
    print("列表没有第 6 个元素")

# 4. TypeError：类型不匹配
try:
    result = "hello" + 123
except TypeError as e:
    print(f"类型错误：{e}")

# 5. ValueError：值不合法
try:
    number = int("abc")
except ValueError as e:
    print(f"值错误：{e}")

# 6. TimeoutError：请求超时（requests 里常用）
import requests
try:
    response = requests.get("https://slow.example.com", timeout=5)
except requests.exceptions.Timeout:
    print("请求超时")
except requests.exceptions.ConnectionError:
    print("连接失败")

测试脚本里，接口调用必须加异常处理（超时、连接失败、5xx）。不加的话，一条接口超时整个测试脚本就崩了，后面的用例全不跑。加 try/except 后，单条失败不影响其他用例。

案例：读配置 + 写日志 + 容错执行

把文件操作 + 异常处理串起来，写一个「读配置 → 跑测试 → 写日志 → 容错」的完整脚本：

python — 完整案例
import json
import logging
from datetime import datetime

# 1. 配置日志（写文件 + 控制台）
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler("test_run.log", encoding="utf-8"),
        logging.StreamHandler(),   # 同时打印到控制台
    ]
)
logger = logging.getLogger(__name__)

# 2. 读配置
def load_config(path="config.json"):
    """读配置，出错用默认值"""
    try:
        with open(path, "r", encoding="utf-8") as f:
            config = json.load(f)
        logger.info(f"配置加载成功：{path}")
        return config
    except FileNotFoundError:
        logger.warning(f"配置文件 {path} 不存在，使用默认配置")
        return {"timeout": 10, "retries": 3, "base_url": "https://api.example.com"}
    except json.JSONDecodeError as e:
        logger.error(f"配置 JSON 格式错误：{e}，使用默认配置")
        return {"timeout": 10, "retries": 3, "base_url": "https://api.example.com"}

# 3. 跑测试（模拟）
def run_test_case(test_name, base_url, timeout, retries):
    """模拟跑一条用例（带重试）"""
    for attempt in range(1, retries + 1):
        try:
            # 模拟接口调用
            logger.info(f"[{test_name}] 第 {attempt} 次调用 {base_url}")
            # 假设 70% 概率成功
            if attempt >= 2:
                logger.info(f"[{test_name}] 通过")
                return True
            else:
                logger.warning(f"[{test_name}] 第 {attempt} 次失败，重试")
        except Exception as e:
            logger.error(f"[{test_name}] 异常：{e}")

    logger.error(f"[{test_name}] 重试 {retries} 次后仍失败")
    return False

# 4. 主流程
def main():
    config = load_config()
    logger.info(f"开始测试，配置：{config}")

    test_cases = [
        "用户登录",
        "创建订单",
        "支付订单",
        "查询订单",
    ]

    results = {}
    for test_name in test_cases:
        passed = run_test_case(
            test_name,
            config["base_url"],
            config["timeout"],
            config["retries"]
        )
        results[test_name] = "通过" if passed else "失败"

    # 5. 写结果到文件
    report_path = f"report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
    try:
        with open(report_path, "w", encoding="utf-8") as f:
            f.write(f"测试报告 - {datetime.now().isoformat()}\n")
            f.write("=" * 40 + "\n")
            for name, result in results.items():
                f.write(f"{name}: {result}\n")
            passed_count = sum(1 for r in results.values() if r == "通过")
            f.write(f"\n总计：{len(results)}  通过：{passed_count}  失败：{len(results) - passed_count}\n")
        logger.info(f"报告已写入：{report_path}")
    except IOError as e:
        logger.error(f"写报告失败：{e}")

    logger.info("测试完成")

if __name__ == "__main__":
    main()

测试脚本的「容错三件套」：① 读文件用 with + try/except；② 接口调用加 超时 + 重试；③ 写日志用 logging 模块（别用 print，日志要能写文件、带时间戳、分级别）。有了这三样，脚本崩不了、问题查得到。
