pywinauto 的两种后端

Windows 有两套「控件访问」API，pywinauto 都封装了：

两种后端对比
# backend="win32"（默认）
# - 速度快
# - 只能访问标准 Win32 控件（按钮、文本框、列表框）
# - 不能访问 UIA 控件（WPF、WinUI）
# - 适合老应用（记事本、计算器）

# backend="uia"
# - 慢一点
# - 能访问所有控件（包括 UIA、WPF、WinUI）
# - 适合现代应用（Office、VS Code、Electron）

# 选择策略：
# - 老 Win32 应用 → win32（快）
# - 现代应用（WPF/WinUI）→ uia（全）
# - 不确定 → 两个都试

窗口定位：找到目标应用

python — 窗口定位
from pywinauto import Application, Desktop
import subprocess
import time

# 1. 启动应用（从脚本里启动，最稳）
app = Application(backend="uia").start("notepad.exe")
notepad_window = app.window(title="无标题 - 记事本")
notepad_window.wait("visible", timeout=5)   # 等窗口可见
print(f"找到记事本：{notepad_window.window_text()}")

# 2. 找已打开的窗口（不启动）
desktop = Desktop(backend="uia")
notepad = desktop.window(title="无标题 - 记事本")
if notepad.exists(timeout=3):
    print(f"记事本已打开：{notepad.window_text()}")

# 3. 按类名找窗口（更精准）
notepad = app.window(title_re=".*记事本", class_name="Notepad")
# title_re：正则匹配标题
# class_name：Windows 窗口类名

# 4. 列出所有顶层窗口（调试用）
for window in Desktop(backend="uia").windows():
    print(f"  {window.window_text()}（{window.class_name()}）")

控件定位：找按钮、文本框

python — 控件定位
from pywinauto import Application

app = Application(backend="uia").start("notepad.exe")
notepad = app.window(title_re=".*记事本")

# 按类型找控件（最常用）
# .child_window(...) 找直接子控件
# .descendants(...) 找所有后代控件（深度搜索）

# 找按钮（按名字）
save_btn = notepad.child_window(control_type="Button", title="保存")
if save_btn.exists(timeout=3):
    save_btn.click()
    print("保存按钮已点击")

# 找编辑框
edit_box = notepad.child_window(control_type="Edit")
edit_box.set_edit_text("Hello, pywinauto!")
print(f"编辑框内容：{edit_box.window_text()}")

# 找菜单项
menu = notepad.child_window(control_type="MenuItem", title="文件")
menu.select_item("新建")

# 按 auto_id 找（WPF/WinUI 应用常用）
# auto_id 是控件的"程序化 ID"（比 title 稳定）
# 用 Inspect.exe 工具查看 auto_id
btn = notepad.child_window(auto_id="SaveButton")
btn.click()

# 遍历所有控件（调试）
for child in notepad.descendants():
    print(f"  控件：{child.control_type} 名字：{child.window_text()}")

控件定位「三件套」：control_type（按钮/文本框/菜单）、title（显示文字）、auto_id（程序化 ID，最稳定）。WPF/WinUI 应用优先用 auto_id，老 Win32 应用用 title。用 Inspect.exe（Windows SDK 自带）查看控件属性。

常用操作：点击、输入、断言

python — 控件操作
from pywinauto import Application
import time

app = Application(backend="uia").start("notepad.exe")
notepad = app.window(title_re=".*记事本")

# 1. 点击
save_btn = notepad.child_window(control_type="Button", title="保存")
save_btn.click()                 # 左键
save_btn.click(button="right")  # 右键
save_btn.click(double=True)     # 双击
save_btn.set_focus()            # 获取焦点

# 2. 输入（Edit 控件）
edit_box = notepad.child_window(control_type="Edit")
edit_box.set_edit_text("Hello, world!")          # 设置整个内容
edit_box.type_keys("new line")                   # 追加输入
edit_box.type_keys("ctrl+a")                     # 组合键

# 3. 断言
# 控件存在
assert notepad.exists(timeout=5), "记事本没打开"

# 控件属性
assert save_btn.is_enabled(), "保存按钮被禁用"
assert save_btn.is_visible(), "保存按钮不可见"

# 文本值
assert edit_box.window_text() == "Hello, world!"

# 等待条件（最常用）
notepad.wait("visible", timeout=10)
save_btn.wait("enabled", timeout=5)
edit_box.wait("exists", timeout=3)

# 4. 选择（ComboBox / ListBox）
combo = notepad.child_window(control_type="ComboBox", title_re=".*字体.*")
combo.select_item("等线")

list_box = notepad.child_window(control_type="ListBox")
list_box.select_item("第三项")

# 5. 拖动（拖拽文件/滑块）
# 先定位两个控件的位置
from pywinauto import mouse
source_rect = notepad.rectangle()
target_x = source_rect.left + 100
target_y = source_rect.top + 50
mouse.click(button="left")
mouse.move(target_x, target_y, drag=True)
mouse.click(button="right")

两种后端的差异

python — 后端差异
# win32 后端（老应用，快）
app_win32 = Application(backend="win32").start("notepad.exe")
notepad_win32 = app_win32.top_window()
# win32 用 .title 找控件（不是 .child_window）
save_btn_win32 = notepad_win32.window(title="保存")
save_btn_win32.click()

# uia 后端（现代应用，全）
app_uia = Application(backend="uia").start("notepad.exe")
notepad_uia = app_uia.window(title_re=".*记事本")
save_btn_uia = notepad_uia.child_window(control_type="Button", title="保存")
save_btn_uia.click()

# 调试：dump 控件树
notepad_uia.print_control_identifiers()
# 输出每个控件的 auto_id、control_type、title、enable 状态
# 帮你"找控件"时确定参数

pywinauto 「找控件」三步走：① dump 控件树（print_control_identifiers()）；② 确定 control_type + title/auto_id；③ 用 child_window(...) 定位。dump 是「调试神器」，不知道控件叫什么名字时全用它。

案例：自动操作记事本 + 验证

把上面学到的串起来，写一个「打开记事本 → 输入文本 → 保存 → 验证」的完整测试：

python — 完整案例
from pywinauto import Application
import pytest

class NotepadTest:
    """记事本自动化测试（pywinauto）"""

    @pytest.fixture(autouse=True)
    def setup(self):
        """启动记事本 + 清理"""
        self.app = Application(backend="uia").start("notepad.exe")
        self.notepad = self.app.window(title_re=".*记事本")
        self.notepad.wait("visible", timeout=5)
        yield
        self.notepad.close()

    def test_input_text(self):
        """输入文本 + 验证"""
        edit_box = self.notepad.child_window(control_type="Edit")
        edit_box.set_edit_text("Hello, pywinauto!")

        # 验证内容
        assert edit_box.window_text() == "Hello, pywinauto!"

    def test_save_file(self):
        """保存文件 + 验证"""
        import os

        # 输入文本
        edit_box = self.notepad.child_window(control_type="Edit")
        edit_box.set_edit_text("测试保存内容")

        # 保存（Ctrl+S）
        file_path = os.path.join(os.path.expanduser("~"), "Desktop", "test_notepad.txt")
        # 用菜单"文件 → 另存为"（更可控）
        menu = self.notepad.child_window(control_type="MenuItem", title="文件")
        menu.select_item("另存为")

        # 保存对话框里输入路径
        time.sleep(1)
        save_dialog = self.app.window(title_re=".*保存.*")
        save_dialog.wait("visible", timeout=5)
        edit_filename = save_dialog.child_window(control_type="Edit", auto_id="1001")
        edit_filename.set_edit_text(file_path)

        # 点击"保存"
        save_btn = save_dialog.child_window(control_type="Button", title="保存(S)")
        save_btn.click()

        # 验证文件存在
        time.sleep(1)
        assert os.path.exists(file_path), "保存文件不存在"
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
        assert content == "测试保存内容", "文件内容不对"

# 运行
# pytest test_notepad.py -v -s

pywinauto 比 pyautogui「稳」在哪：找控件树里的控件，不靠坐标/截图。界面「位置变了」不影响（控件还在），「改名了」才影响（title 变了）。WPF/WinUI 应用用 auto_id（程序化 ID，最稳定），老 Win32 应用用 title。配合 pytest fixture 自动启动/关闭应用，一条用例一个「干净环境」。
