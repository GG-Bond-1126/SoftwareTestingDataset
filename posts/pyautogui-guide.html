鼠标操作：点击、拖拽、滚动

python — 鼠标操作
import pyautogui
import time

# 点击
pyautogui.click()                     # 当前位置左键点击
pyautogui.click(200, 300)            # 移到 (200, 300) 再点
pyautogui.rightClick(200, 300)       # 右键
pyautogui.middleClick()              # 中键
pyautogui.doubleClick(200, 300)      # 双击
pyautogui.tripleClick(200, 300)     # 三击（选中一行文本）

# 拖拽
pyautogui.dragTo(500, 400, duration=0.5)   # 拖到 (500, 400)，0.5 秒内完成
pyautogui.dragRel(100, 50, duration=0.3)   # 相对位置拖（向右 100，向下 50）

# 滚动
pyautogui.scroll(-3)                 # 向下滚 3 格
pyautogui.scroll(5)                  # 向上滚 5 格
pyautogui.scrollTo(100, 50)         # 滚到绝对位置

# 移动
pyautogui.moveTo(300, 400)          # 移动鼠标（不点击）
pyautogui.moveTo(300, 400, duration=1)   # 1 秒内移动（模拟"人"）

# 测试场景：模拟"用户操作"
# 1. 滚动到页面底部
pyautogui.scroll(-100)
time.sleep(0.5)

# 2. 点击"提交"按钮（坐标已提前截图标注）
pyautogui.click(750, 600)

# 3. 拖拽滑块验证
pyautogui.moveTo(100, 300)
pyautogui.dragTo(500, 300, duration=0.8)

pyautogui 的坐标是屏幕绝对坐标（左上角 (0, 0)）。测试前必须「截图标注坐标」，脚本里用这些坐标。屏幕分辨率变了坐标就全错，所以 PC 自动化脚本「换机器基本要重新标坐标」。

图像识别：找图 + 点击

PC 自动化最常用的「找图」操作：截一个「按钮图片」，在屏幕上找它，找到了就点。

python — 图像识别
import pyautogui
import time

# 1. 先截"目标图片"保存为文件
# （比如截一个"保存"按钮，保存为 save_btn.png）

# 2. 在屏幕上找这个图片
location = pyautogui.locateOnScreen("save_btn.png", confidence=0.9)
if location:
    center = pyautogui.center(location)
    print(f"找到图片：{center}")
    pyautogui.click(center)
    print("已点击")
else:
    print("没找到图片（可能界面变了）")

# 3. 找多张图片（哪个先找到用哪个）
for image_name in ["save_btn_v2.png", "save_btn.png"]:
    location = pyautogui.locateOnScreen(image_name, confidence=0.85)
    if location:
        pyautogui.click(pyautogui.center(location))
        break

# 4. 等待图片出现（带超时）
import time

def wait_for_image(image_path, timeout=10, interval=0.5):
    """等待图片出现，最多等 timeout 秒。"""
    start = time.time()
    while time.time() - start < timeout:
        location = pyautogui.locateOnScreen(image_path, confidence=0.85)
        if location:
            return pyautogui.center(location)
        time.sleep(interval)
    return None   # 超时没找到

location = wait_for_image("success_popup.png", timeout=15)
if location:
    print("成功弹窗已出现")
    pyautogui.click(location)
else:
    print("❌ 成功弹窗没出现（15 秒超时）")

# 5. 灰度匹配（黑白对比更快更稳）
import cv2
import numpy as np

def find_image_gray(screen_path, target_path):
    """灰度图像识别（比彩色快 3 倍）"""
    screen = cv2.imread(screen_path)
    target = cv2.imread(target_path)

    # 转灰度
    screen_gray = cv2.cvtColor(screen, cv2.COLOR_BGR2GRAY)
    target_gray = cv2.cvtColor(target, cv2.COLOR_BGR2GRAY)

    # 模板匹配
    result = cv2.matchTemplate(screen_gray, target_gray, cv2.TM_CCOEFF_NORMED)
    _, max_value, _, max_loc = cv2.minMaxLoc(result)

    if max_value > 0.8:   # 置信度 > 80%
        x, y = max_loc
        return (x + target_gray.shape[1] // 2, y + target_gray.shape[0] // 2)
    return None

安全机制：别让脚本「失控」

python — 安全机制
import pyautogui

# 1. Failsafe（紧急停止）
# 鼠标移到屏幕左上角 (0, 0) 立即抛出异常
pyautogui.FAILSAFE = True
try:
    pyautogui.moveTo(500, 300)
except pyautogui.FailSafeException:
    print("用户紧急停止了脚本（鼠标移到左上角）")

# 2. Pause（操作间隔）
# 每次操作后自动等 0.5s（给界面反应时间）
pyautogui.PAUSE = 0.5
pyautogui.press("enter")    # 自动等 0.5s
pyautogui.press("enter")    # 自动等 0.5s

# 3. Speed（移动速度）
# duration 参数控制移动速度（模拟"人"）
pyautogui.moveTo(500, 300, duration=1.0)   # 1 秒内移动

# 4. 屏幕分辨率检测（换机器前检查）
screen_width, screen_height = pyautogui.size()
print(f"屏幕：{screen_width}x{screen_height}")
if screen_width < 1920:
    print("⚠️ 屏幕分辨率低于 1920，坐标可能偏移")

案例：自动操作 Excel + 截图验证

把 pyautogui 串起来，写一个「打开 Excel → 填写数据 → 保存 → 截图验证」的完整案例：

python — 完整案例
import pyautogui
import pyperclip
import subprocess
import time
import os
from datetime import datetime

pyautogui.PAUSE = 0.3   # 每次操作后等 0.3s
pyautogui.FAILSAFE = True

# 1. 打开 Excel（假设已安装）
subprocess.Popen(["excel.exe"])
time.sleep(3)   # 等 Excel 启动

# 2. 新建工作簿（Ctrl+N）
pyautogui.hotkey("ctrl", "n")
time.sleep(1)

# 3. 填写数据（A1 → B1 → C1）
# 数据（用剪贴板输入，支持中文）
data = [
    ("用户", "订单号", "金额"),
    ("张三", "ORD-001", "99.99"),
    ("李四", "ORD-002", "199.98"),
]

row = 0
for values in data:
    for col, value in enumerate(values):
        # 移到 A1 + 偏移
        x = 150 + col * 200   # 列坐标（假设每列 200px）
        y = 250 + row * 30     # 行坐标（假设每行 30px）
        pyautogui.click(x, y)
        time.sleep(0.2)

        # 写入剪贴板 + 粘贴（支持中文）
        pyperclip.copy(str(value))
        pyautogui.hotkey("ctrl", "v")
        time.sleep(0.2)

    row += 1

# 4. 保存（Ctrl+S）
pyautogui.hotkey("ctrl", "s")
time.sleep(1)

# 5. 输入文件名
pyperclip.copy("test_data.xlsx")
pyautogui.hotkey("ctrl", "v")
pyautogui.press("enter")
time.sleep(1)

# 6. 截图验证
timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
screenshot_path = f"excel_test_{timestamp}.png"
pyautogui.screenshot(screenshot_path)
print(f"截图已保存：{screenshot_path}")

# 7. 验证文件存在
desktop = os.path.join(os.path.expanduser("~"), "Desktop")
saved_files = [f for f in os.listdir(desktop) if f == "test_data.xlsx"]
if saved_files:
    print("✅ Excel 保存成功")
else:
    print("⚠️ 文件可能保存在其他位置")

# 8. 关闭 Excel
pyautogui.hotkey("ctrl", "w")   # 关闭当前工作簿
time.sleep(0.5)

PC 自动化「五步法」：截图标注坐标 → pyautogui 模拟操作 → 等待界面响应 → 截图验证 → 安全机制（Failsafe + Pause）。pyautogui 的最大优势是「简单」（一行代码模拟点击/输入），最大劣势是「脆弱」（分辨率变了、界面改一点就失效）。适合「界面稳定、操作固定」的场景。
