多级降级：主策略失败试备用策略

「控件树定位」失败了，试「图像识别」；图像识别也失败，试「坐标点击」。逐级降级，保证「总能定位到」：

python — 多级降级
import pyautogui
import uiautomation as auto
import time

def locate_button_multi_level(
    control_name,          # 控件名（uiautomation 用）
    image_path,           # 参考图（图像识别用）
    fallback_coord,       # 兜底坐标（最脆弱）
    timeout=5,
):
    """多级降级定位按钮。

    策略顺序：
    1. uiautomation 控件树（最精准）
    2. 图像识别（中等）
    3. 坐标点击（最脆弱）

    Returns:
        定位成功返回 (x, y)，全失败返回 None
    """
    # 策略 1：控件树（最精准）
    try:
        window = auto.WindowControl(searchDepth=1, subTitle="目标应用")
        if window.Exists(1):
            btn = window.child_window(control_type="Button", title=control_name)
            if btn.Exists(timeout):
                location = btn.rectangle()
                center = ((location.left + location.right) // 2,
                          (location.top + location.bottom) // 2)
                print(f"  控件树定位成功：{center}")
                return center
    except Exception:
        pass

    # 策略 2：图像识别（中等）
    location = pyautogui.locateOnScreen(image_path, confidence=0.85)
    if location:
        center = pyautogui.center(location)
        print(f"  图像识别定位成功：{center}")
        return center

    # 策略 3：坐标点击（最脆弱，兜底）
    if fallback_coord:
        print(f"  坐标兜底定位：{fallback_coord}")
        return fallback_coord

    # 全失败
    print(f"  ❌ 三种策略都失败：{control_name}")
    return None

# 使用
location = locate_button_multi_level(
    control_name="保存",
    image_path="buttons/save_btn.png",
    fallback_coord=(1420, 80)   # 从 1920x1080 屏幕截的坐标
)
if location:
    pyautogui.click(location)
    print("保存按钮已点击")

重试机制：操作失败重试 N 次

python — 重试装饰器
import functools
import time
import logging

logger = logging.getLogger(__name__)

def auto_retry(max_retries=3, delay=2.0, retry_on=None):
    """自动重试装饰器（PC 操作偶发失败时用）。

    Args:
        max_retries: 最大重试次数
        delay: 每次重试间隔（秒）
        retry_on: 要重试的异常类型（默认 Exception）
    """
    if retry_on is None:
        retry_on = Exception

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            last_error = None
            for attempt in range(1, max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except retry_on as e:
                    last_error = e
                    logger.warning(f"  {func.__name__} 第 {attempt} 次失败：{e}")
                    if attempt < max_retries:
                        time.sleep(delay)
            logger.error(f"  {func.__name__} 重试 {max_retries} 次后仍失败")
            raise last_error
        return wrapper
    return decorator

# 使用：鼠标点击重试（界面没加载完时点击可能"点空"）
@auto_retry(max_retries=3, delay=1.0)
def click_button(image_path):
    """点击按钮（失败重试 3 次）。"""
    location = pyautogui.locateOnScreen(image_path, confidence=0.9)
    if location is None:
        raise RuntimeError(f"没找到 {image_path}")
    pyautogui.click(pyautogui.center(location))

# 调用
click_button("buttons/confirm_btn.png")
# 没找到 → 重试 3 次（每次等 1s）→ 还失败才报错

超时配置：每个操作设「最长等待」

python — 超时配置
import pyautogui
import time

# 不同操作设不同超时（不是"统一 10s"）
TIMEOUTS = {
    "window_start": 15,    # 启动应用（可能慢）
    "element_visible": 10, # 元素可见（中等）
    "api_response": 30,    # 接口响应（网络慢）
    "file_generation": 60, # 文件生成（可能很慢）
    "click_response": 5,   # 点击响应（应该快）
}

def wait_with_timeout(condition_func, timeout_key):
    """按"操作类型"设超时的等待。"""
    timeout = TIMEOUTS[timeout_key]
    start = time.time()
    while time.time() - start < timeout:
        try:
            if condition_func():
                elapsed = time.time() - start
                print(f"  {timeout_key} 成功（{elapsed:.1f}s / {timeout}s）")
                return True
        except Exception:
            pass
        time.sleep(0.5)
    print(f"  ❌ {timeout_key} 超时（{timeout}s）")
    return False

# 使用
def window_started():
    import uiautomation as auto
    return auto.WindowControl(searchDepth=1, subTitle="记事本").Exists(1)

if not wait_with_timeout(window_started, "window_start"):
    print("记事本启动失败")
    exit(1)

界面改版的应对策略

应对策略
# 1. 参考图版本管理（v1/v2/v3）
# 界面改版后，"旧参考图"匹配不上
# 维护多版本参考图，按顺序尝试
def locate_versioned(image_paths):
    """按版本找图（v3 优先，v1 兜底）。"""
    for path in image_paths:   # ["btn_v3.png", "btn_v2.png", "btn_v1.png"]
        location = pyautogui.locateOnScreen(path, confidence=0.85)
        if location:
            print(f"  找到：{path}")
            return pyautogui.center(location)
    return None

# 2. 控件树 + 图像识别双备份
# 控件树定位失败（控件改名），图像识别兜底
def locate_dual(control_name, image_path, fallback_coord=None):
    """控件树 + 图像识别双备份。"""
    # 控件树（优先）
    import uiautomation as auto
    try:
        btn = auto.WindowControl(searchDepth=1, subTitle="目标").child_window(
            control_type="Button", title=control_name
        )
        if btn.Exists(3):
            return btn.rectangle().center()
    except Exception:
        pass

    # 图像识别（兜底）
    location = pyautogui.locateOnScreen(image_path, confidence=0.85)
    if location:
        return pyautogui.center(location)

    # 坐标（最后兜底）
    return fallback_coord

#     "fallback_coord": [1420, 80]
#   }

PC 自动化「稳定性三原则」：① 多级降级（控件树 → 图像识别 → 坐标，逐级兜底）；② 重试机制（偶发失败重试 3 次，不是「一次失败就崩」）；③ 超时配置（不同操作不同超时，不是「统一 10s」）。界面改版时改配置，不改代码。

# 多级降级定位
def locate_control(config_item, timeout=5):
    """多级降级定位控件（控件树 → 图像识别 → 坐标）。"""
    # 策略 1：控件树
    try:
        ctrl = auto.WindowControl(searchDepth=1, subTitle="后台管理系统")
        if ctrl.Exists(1):
            target = ctrl.child_window(
                control_type="Edit" if "input" in config_item["control_name"] else "Button",
                title=config_item["control_name"],
            )
            if target.Exists(timeout):
                rect = target.rectangle()
                center = ((rect.left + rect.right) // 2, (rect.top + rect.bottom) // 2)
                logger.info(f"  控件树定位：{center}")
                return center
    except Exception:
        pass

    # 策略 2：图像识别
    location = pyautogui.locateOnScreen(config_item["image_path"], confidence=0.85)
    if location:
        center = pyautogui.center(location)
        logger.info(f"  图像识别定位：{center}")
        return center

    # 策略 3：坐标兜底
    if config_item["fallback_coord"]:
        logger.info(f"  坐标兜底：{config_item['fallback_coord']}")
        return tuple(config_item["fallback_coord"])

    logger.error(f"  ❌ 定位失败：{config_item['control_name']}")
    return None

# 重试装饰器
def auto_retry(max_retries=3, delay=1.0):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    logger.warning(f"  {func.__name__} 第 {attempt} 次失败：{e}")
                    if attempt < max_retries:
                        time.sleep(delay)
            raise
        return wrapper
    return decorator

PC 自动化「稳定四件套」：多级降级（控件树 → 图像识别 → 坐标）、重试机制（偶发失败重试 N 次）、超时配置（不同操作不同超时）、配置化（坐标/图片路径放 config.json，界面改版改配置不改代码）。四件套配齐，PC 自动化脚本「界面改一点也不崩」。
