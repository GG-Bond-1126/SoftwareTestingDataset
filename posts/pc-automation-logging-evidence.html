操作日志：记录每一步「做了什么」

PC 自动化脚本「跑了 100 步」，失败后「翻日志」就能定位「第几步挂的」。核心是「每一步操作都打日志」：

python — 操作日志
import logging
from datetime import datetime
from logging.handlers import RotatingFileHandler

def setup_pc_test_logger(test_name="pc_test"):
    """配置 PC 测试日志（控制台 + 文件双通道）。"""
    logger = logging.getLogger(f"pc_{test_name}")
    logger.setLevel(logging.DEBUG)

    # 控制台（INFO 级别）
    console = logging.StreamHandler()
    console.setLevel(logging.INFO)
    console.setFormatter(logging.Formatter("[%(levelname)s] %(message)s"))
    logger.addHandler(console)

    # 文件（DEBUG 级别，按大小轮转）
    log_file = f"logs/{test_name}_{datetime.now().strftime('%Y%m%d')}.log"
    import os
    os.makedirs("logs", exist_ok=True)
    file_handler = RotatingFileHandler(
        log_file, maxBytes=5*1024*1024, backupCount=3, encoding="utf-8"
    )
    file_handler.setLevel(logging.DEBUG)
    file_handler.setFormatter(logging.Formatter(
        "%(asctime)s [%(levelname)s] %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    ))
    logger.addHandler(file_handler)

    return logger

# 使用：每一步操作都打日志
logger = setup_pc_test_logger("excel_test")

def logged_click(x, y, description):
    """点击 + 打日志。"""
    logger.info(f"点击 ({x}, {y})：{description}")
    pyautogui.click(x, y)
    logger.debug(f"  点击完成")

def logged_type(text, description):
    """输入 + 打日志。"""
    logger.info(f"输入：{text}（{description}）")
    pyperclip.copy(text)
    pyautogui.hotkey("ctrl", "v")

# 测试流程（每一步都打日志）
logger.info("=== 测试开始 ===")
logged_click(500, 300, "点击'新建'按钮")
logged_type("test_data", "输入测试数据")
logged_click(800, 600, "点击'保存'按钮")
logger.info("=== 测试结束 ===")

# 失败后翻日志（定位"哪一步"）
# 2026-11-23 10:00:00 [INFO] 点击 (500, 300)：点击'新建'按钮
# 2026-11-23 10:00:01 [INFO] 输入：test_data（输入测试数据）
# 2026-11-23 10:00:02 [ERROR] 点击 (800, 600)：点击'保存'按钮 → 失败

自动截图：关键节点「拍照」

PC 自动化「取证」最直观的方式是截图。关键节点自动截图，失败后「看截图」就知道「界面长什么样」：

python — 自动截图
import pyautogui
import os
import logging
from datetime import datetime

class ScreenshotManager:
    """截图管理器（自动命名 + 归档）。"""

    def __init__(self, screenshot_dir="screenshots", keep_last=50):
        self.dir = screenshot_dir
        self.keep_last = keep_last
        os.makedirs(screenshot_dir, exist_ok=True)

    def take(self, description, prefix=""):
        """截图（自动命名 + 记录日志）。

        Args:
            description: 描述（"点击保存前"、"保存成功后"）
            prefix: 前缀（"excel"、"game"）

        Returns:
            截图路径
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        filename = f"{prefix}_{description.replace(' ', '_')}_{timestamp}.png"
        path = os.path.join(self.dir, filename)

        pyautogui.screenshot(path)
        logging.getLogger("pc_test").info(f"截图：{path}")

        # 清理旧截图（保留最近 N 张）
        self._cleanup()
        return path

    def take_on_failure(self, test_name):
        """失败时截图（自动命名 + 标记）。"""
        return self.take(f"FAIL_{test_name}", prefix="error")

    def _cleanup(self):
        """清理旧截图。"""
        files = sorted(os.listdir(self.dir))
        old_files = files[:-self.keep_last] if len(files) > self.keep_last else []
        for old in old_files:
            os.remove(os.path.join(self.dir, old))

# 使用
shots = ScreenshotManager(keep_last=50)

# 关键节点截图
shots.take("测试开始", prefix="excel")
shots.take("输入数据后", prefix="excel")
shots.take("保存成功", prefix="excel")

# 失败时截图
try:
    # 测试逻辑...
    pass
except Exception:
    shots.take_on_failure("save_button")
    raise

失败取证：截图 + 日志 + 异常栈

python — 失败取证
import pyautogui
import traceback
import logging
from datetime import datetime

class EvidenceCollector:
    """失败取证（截图 + 日志 + 异常栈）。"""

    def __init__(self, evidence_dir="evidence"):
        self.dir = evidence_dir
        import os
        os.makedirs(self.dir, exist_ok=True)
        self.logger = logging.getLogger("evidence")

    def collect(self, test_name, exception=None):
        """收集失败证据。

        Args:
            test_name: 测试名称
            exception: 异常对象（有就记异常栈）

        Returns:
            证据目录路径
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        evidence_path = os.path.join(self.dir, f"{test_name}_{timestamp}")
        os.makedirs(evidence_path, exist_ok=True)

        # 1. 截图
        screenshot_path = os.path.join(evidence_path, "screenshot.png")
        pyautogui.screenshot(screenshot_path)
        self.logger.info(f"截图已保存：{screenshot_path}")

        # 2. 日志（导出当前日志文件）
        import shutil
        log_files = [f for f in os.listdir("logs") if test_name in f]
        if log_files:
            source_log = os.path.join("logs", log_files[-1])
            target_log = os.path.join(evidence_path, "test.log")
            shutil.copy2(source_log, target_log)
            self.logger.info(f"日志已导出：{target_log}")

        # 3. 异常栈
        if exception:
            traceback_path = os.path.join(evidence_path, "traceback.txt")
            with open(traceback_path, "w", encoding="utf-8") as f:
                f.write(f"测试：{test_name}\n")
                f.write(f"时间：{datetime.now().isoformat()}\n")
                f.write(f"异常：{type(exception).__name__}: {exception}\n")
                f.write("=" * 50 + "\n")
                f.write("".join(traceback.format_exception(type(exception), exception, exception.__traceback__)))
            self.logger.info(f"异常栈已保存：{traceback_path}")

        # 4. 生成证据清单
        manifest_path = os.path.join(evidence_path, "manifest.txt")
        with open(manifest_path, "w", encoding="utf-8") as f:
            f.write(f"测试名称：{test_name}\n")
            f.write(f"失败时间：{datetime.now().isoformat()}\n")
            f.write(f"证据目录：{evidence_path}\n")
            f.write(f"包含文件：{os.listdir(evidence_path)}\n")

        self.logger.info(f"=== 失败证据已收集：{evidence_path} ===")
        return evidence_path

# 使用
collector = EvidenceCollector()

try:
    # 测试逻辑
    run_test("test_excel")
except Exception as e:
    evidence_path = collector.collect("test_excel", exception=e)
    print(f"测试失败，证据已保存：{evidence_path}")
    raise

日志归档：保留最近 N 次

python — 日志归档
import os
import shutil
import zipfile
import glob
from datetime import datetime

class LogArchiver:
    """日志归档（打包 + 保留最近 N 次）。"""

    def __init__(self, log_dir="logs", keep_last=30):
        self.log_dir = log_dir
        self.keep_last = keep_last

    def archive(self, test_name, date_tag):
        """归档一次测试的日志（打包 zip）。"""
        # 找该测试的日志文件
        log_files = glob.glob(os.path.join(self.log_dir, f"{test_name}_*.log"))
        if not log_files:
            print(f"没找到日志：{test_name}")
            return None

        # 打包 zip
        zip_name = f"{test_name}_{date_tag}.zip"
        zip_path = os.path.join(self.log_dir, "archive", zip_name)
        os.makedirs(os.path.dirname(zip_path), exist_ok=True)

        with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as zf:
            for log_file in log_files:
                zf.write(log_file, os.path.basename(log_file))
            # 也打包截图
            screenshots = glob.glob(os.path.join("screenshots", f"{test_name}_*.png"))
            for shot in screenshots:
                zf.write(shot, os.path.join("screenshots", os.path.basename(shot)))

        print(f"日志已归档：{zip_path}")
        return zip_path

    def cleanup(self):
        """清理旧归档（保留最近 N 个）。"""
        archives = sorted(glob.glob(os.path.join(self.log_dir, "archive", "*.zip")))
        old_archives = archives[:-self.keep_last] if len(archives) > self.keep_last else []
        for old in old_archives:
            os.remove(old)
        print(f"旧归档已清理（保留最近 {self.keep_last} 个）")

# 使用
archiver = LogArchiver(keep_last=30)

# 测试完成后归档
archiver.archive("excel_test", "20261123")

# 定期清理
archiver.cleanup()

案例：测试套件自动截图 + 日志归档

把「操作日志 + 自动截图 + 失败取证 + 日志归档」串起来，写一个「测试套件取证」的完整案例：

python — 完整案例
import pytest
import pyautogui
import logging
import os
from datetime import datetime

# pytest fixture：自动截图 + 日志
@pytest.fixture(autouse=True)
def test_evidence(request):
    """每个测试自动截图 + 日志。"""
    test_name = request.node.name

    # 日志
    logger = logging.getLogger(f"test_{test_name}")
    logger.info(f"=== 测试开始：{test_name} ===")

    # 截图（测试前）
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    before_path = f"evidence/{test_name}_before_{timestamp}.png"
    os.makedirs("evidence", exist_ok=True)
    pyautogui.screenshot(before_path)
    logger.info(f"测试前截图：{before_path}")

    yield   # 测试执行

    # 测试后截图
    after_path = f"evidence/{test_name}_after_{timestamp}.png"
    pyautogui.screenshot(after_path)
    logger.info(f"测试后截图：{after_path}")

    # 记录结果
    if hasattr(request.node, "rep_call") and request.node.rep_call.passed:
        logger.info(f"=== 测试通过：{test_name} ===")
    else:
        logger.error(f"=== 测试失败：{test_name} ===")

# pytest 钩子：失败时自动取证
def pytest_runtest_makereport(item, call):
    """测试失败时收集证据。"""
    if call.when == "call" and call.excinfo:
        # 截图
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        fail_path = f"evidence/{item.name}_FAIL_{timestamp}.png"
        os.makedirs("evidence", exist_ok=True)
        pyautogui.screenshot(fail_path)

        # 异常栈
        traceback_path = f"evidence/{item.name}_FAIL_{timestamp}_traceback.txt"
        import traceback
        with open(traceback_path, "w", encoding="utf-8") as f:
            f.write("".join(traceback.format_exception(*call.excinfo)))

        print(f"\n⚠️  测试失败证据：{fail_path}")
        print(f"   异常栈：{traceback_path}")

# 测试用例（自动取证）
def test_open_excel():
    """打开 Excel + 验证。"""
    import subprocess
    import time

    subprocess.Popen(["excel.exe"])
    time.sleep(5)

    # 验证 Excel 窗口
    import uiautomation as auto
    excel_window = auto.WindowControl(searchDepth=1, subTitle="Microsoft Excel")
    assert excel_window.Exists(10), "Excel 窗口没打开"

    # 截图验证
    pyautogui.screenshot("evidence/excel_opened.png")

def test_input_data():
    """输入数据 + 保存。"""
    import pyperclip
    import time

    pyperclip.copy("测试数据 v2")
    pyautogui.hotkey("ctrl", "v")
    time.sleep(0.5)

    pyautogui.hotkey("ctrl", "s")
    time.sleep(1)

PC 自动化「取证四件套」：操作日志（每一步打 log）→ 自动截图（关键节点「拍照」）→ 失败取证（截图 + 日志 + 异常栈打包）→ 日志归档（保留最近 N 次，zip 打包）。脚本「跑挂了」，翻「证据目录」就知道「哪一步挂的 + 当时界面长什么样」。
