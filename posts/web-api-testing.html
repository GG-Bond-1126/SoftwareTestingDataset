LLM 生成接口测试：从 Swagger 文档到可执行 pytest 脚本

环境搭建

本方案用 OpenAI 大模型读取 OpenAPI/Swagger 规范文件，自动生成 pytest + httpx 测试脚本。只需 Python 环境和 LLM API Key，无需浏览器或 Appium。

安装依赖
pip install pytest httpx openai pyyaml

# 如需读取 OpenAPI 规范结构
pip install openapi-spec-validator
配置 LLM 密钥（.env）
# .env
OPENAI_API_KEY=sk-xxxxxxxx
# 使用 gpt-4o-mini 即可满足接口测试生成需求，成本低
LLM_MODEL=gpt-4o-mini

# 被测服务地址
API_BASE_URL=http://localhost:5000

OpenAPI 3.0 规范中 paths 是核心字段。LLM 生成测试时只需读取该文件，无需启动被测服务——测试代码生成和实际运行是两步。

基础应用：LLM 读取 Swagger 自动生成 pytest 用例

下面脚本读取 swagger.json，将每个路径的 schema 信息喂给 LLM，让它生成完整的 pytest + httpx 测试文件。

generate_api_tests.py
"""
从 Swagger 文档生成 pytest + httpx 接口测试
"""
import json, os, pathlib, textwrap
from openai import OpenAI

client = OpenAI()  # 自动读取 OPENAI_API_KEY
MODEL = os.environ.get("LLM_MODEL", "gpt-4o-mini")

def extract_paths(spec: dict) -> dict:
    """提取 OpenAPI 路径信息"""
    return {
        path: {
            method: op
            for method, op in path_item.items()
            if isinstance(op, dict) and "responses" in op
        }
        for path, path_item in spec.get("paths", {}).items()
    }

def build_prompt(path: str, methods: dict, base_url: str) -> str:
    return textwrap.dedent(f"""
        你是接口测试专家。根据以下 OpenAPI 路径定义，生成 pytest + httpx 测试代码。

        路径: {path}
        方法详情:
        {json.dumps(methods, indent=2, ensure_ascii=False)}

        要求：
        1. 使用 httpx.Client 发请求，base_url = "{base_url}"
        2. 对每个方法写正向用例（预期状态码）和异常用例（如 404）
        3. 对 requestBody 必填字段写缺失测试
        4. 使用 pytest fixture 管理 httpx client
        5. 每个测试函数名格式：test_{{method}}_{{path_slug}}_{{case}}
        6. 输出纯 Python 代码，不要 markdown 包裹
    """)

def generate_test_file(spec: dict, base_url: str, out_path: str) -> str:
    paths = extract_paths(spec)
    all_code = []

    header = textwrap.dedent(f"""
        import httpx, pytest

        @pytest.fixture
        def client():
            with httpx.Client(base_url="{base_url}", timeout=10) as c:
                yield c
    """)
    all_code.append(header)

    for path, methods in paths.items():
        prompt = build_prompt(path, methods, base_url)
        resp = client.chat.completions.create(
            model=MODEL,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.1,  # 低温度，代码更稳定
            max_tokens=2048,
        )
        code = resp.choices[0].message.content.strip()
        # 去掉可能的 markdown 包裹
        code = code.removeprefix("```python").removeprefix("```").removesuffix("```").strip()
        all_code.append(code)
        print(f"  生成 {path} 测试 ({len(methods)} 个方法)")

    full_code = "\n\n".join(all_code)
    pathlib.Path(out_path).write_text(full_code, encoding="utf-8")
    return full_code

if __name__ == "__main__":
    spec = json.load(open("swagger.json"))
    base_url = os.environ.get("API_BASE_URL", "http://localhost:5000")
    generate_test_file(spec, base_url, "test_api_generated.py")
    print("生成完成，运行: pytest test_api_generated.py -v")

@pytest.fixture
def client():
    with httpx.Client(base_url="http://localhost:5000", timeout=10) as c:
        yield c

# ── GET /tasks ──
def test_get_tasks_200(client):
    r = client.get("/tasks")
    assert r.status_code == 200
    assert isinstance(r.json(), list)

def test_get_tasks_missing_auth(client):
    r = client.get("/tasks", headers={"Authorization": ""})
    assert r.status_code in [200, 401]

# ── POST /tasks ──
def test_post_tasks_valid(client):
    r = client.post("/tasks", json={"title": "Test", "priority": 3})
    assert r.status_code == 201

def test_post_tasks_missing_title(client):
    r = client.post("/tasks", json={"priority": 1})
    assert r.status_code == 400

def test_post_tasks_invalid_priority(client):
    r = client.post("/tasks", json={"title": "T", "priority": 99})
    assert r.status_code == 400

# ── GET /tasks/{id} ──
def test_get_task_found(client):
    r = client.get("/tasks/1")
    assert r.status_code == 200

def test_get_task_not_found(client):
    r = client.get("/tasks/99999")
    assert r.status_code == 404
LLM 生成代码后务必人工审查一遍，重点关注：请求字段名是否与 schema 一致、状态码断言是否合理、异常分支是否遗漏。

项目进阶：CI 流水线 — 文档变更触发测试再生成

当 Swagger 文档在仓库中发生变更时，CI 自动检测差异，调用 LLM 重新生成测试，运行后与旧版本对比，输出差异报告供人工审查。

detect_doc_change.py — 检测 Swagger 变更
"""
比较当前 swagger.json 与上次提交版本，输出变更摘要
"""
import json, subprocess, sys, pathlib

def diff_spec(old: dict, new: dict) -> list:
    """返回变更摘要列表"""
    changes = []
    old_paths = set(old.get("paths", {}).keys())
    new_paths = set(new.get("paths", {}).keys())

    for p in new_paths - old_paths:
        changes.append(f"+ 新增路径: {p}")
    for p in old_paths - new_paths:
        changes.append(f"- 删除路径: {p}")
    for p in old_paths & new_paths:
        old_methods = set(old["paths"][p].keys())
        new_methods = set(new["paths"][p].keys())
        for m in new_methods - old_methods:
            changes.append(f"~ {p} 新增方法: {m.upper()}")
        for m in old_methods - new_methods:
            changes.append(f"~ {p} 移除方法: {m.upper()}")
    return changes

def main():
    # 读取当前版本和父提交版本
    current = json.load(open("swagger.json"))
    result = subprocess.run(
        ["git", "show", "HEAD~1:swagger.json"],
        capture_output=True, text=True
    )
    if result.returncode != 0:
        print("无历史版本，跳过差异检测")
        sys.exit(0)
    previous = json.loads(result.stdout)

    changes = diff_spec(previous, current)
    if not changes:
        print("Swagger 无变更，跳过重新生成")
        sys.exit(0)

    print("=== Swagger 变更摘要 ===")
    for c in changes:
        print(f"  {c}")
    # 写入文件供后续步骤使用
    pathlib.Path("swagger_changes.txt").write_text("\n".join(changes), encoding="utf-8")

if __name__ == "__main__":
    main()
.github/workflows/api-test-regen.yml
name: API Test Regen on Doc Change
on:
  push:
    branches: [main]
    paths:
      - "swagger.json"
      - "openapi.yaml"

jobs:
  regen:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 2   # 需要父提交做 diff

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install deps
        run: pip install pytest httpx openai pyyaml

      - name: Detect doc change
        id: detect
        run: python detect_doc_change.py || echo "no-change=true" >> $GITHUB_OUTPUT
        continue-on-error: true

      - name: Skip if no change
        if: steps.detect.outputs.no-change == 'true'
        run: echo "Swagger 无变更，跳过"

      - name: Regenerate tests with LLM
        if: steps.detect.outputs.no-change != 'true'
        env:
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
          API_BASE_URL: ${{ vars.API_BASE_URL }}
        run: python generate_api_tests.py

      - name: Run generated tests
        if: steps.detect.outputs.no-change != 'true'
        run: pytest test_api_generated.py -v --tb=short 2>&1 | tee test_output.txt

      - name: Diff old vs new test results
        if: steps.detect.outputs.no-change != 'true'
        run: |
          echo "=== 测试差异报告 ===" > diff_report.md
          echo "变更摘要:" >> diff_report.md
          cat swagger_changes.txt >> diff_report.md
          echo -e "\n测试执行结果:" >> diff_report.md
          tail -20 test_output.txt >> diff_report.md

      - name: Upload diff report
        if: always() && steps.detect.outputs.no-change != 'true'
        uses: actions/upload-artifact@v4
        with:
          name: api-test-diff-report
          path: |
            diff_report.md
            test_api_generated.py
          retention-days: 30

LLM 生成的测试必须经过人工 Code Review 后才能合入主干。建议 CI 中生成测试后先运行 lint 和 type check（python -m mypy），再跑实际测试。
进阶：用 diff -u 对比两次 LLM 生成的测试文件，差异部分高亮显示在 PR 评论中，方便 reviewer 快速定位变更影响。
