三平台核心差异

平台差异对比
# 维度 | Windows | macOS | Linux
# 控件树 | Win32/UIA（丰富） | Accessibility API（可用） | 无（纯图像）
# 鼠标键盘 | pyautogui/pynput | pyautogui/pynput | pyautogui/pynput
# 截图 | pyautogui | pyautogui | pyautogui
# 权限 | 一般不需要 | 需要"辅助功能"权限 | 一般不需要
# 启动应用 | subprocess | open 命令 | xdg-open
# 窗口操作 | pywinauto/uiautomation | AppleScript/pyobjc | xdotool
# 图像识别 | OpenCV（通用） | OpenCV（通用） | OpenCV（通用）

三平台「唯一通用」的工具：pyautogui（鼠标/键盘/截图/图像识别）+ OpenCV（图像匹配）。控件树「平台专属」：Windows 用 pywinauto/uiautomation，Mac 用 AppleScript，Linux 没有。

跨平台工具选择

工具选择
# 全平台通用（推荐）
# - pyautogui：鼠标/键盘/截图/图像识别
# - pynput：监听/控制
# - OpenCV：模板匹配
# - subprocess：启动应用

# Windows 专属
# - pywinauto：控件树（Win32/UIA）
# - uiautomation：控件树遍历
# - pywintypes：Windows API

# macOS 专属
# - AppleScript（osascript）：系统级脚本
# - pyobjc：Python 调 macOS API
# - Accessibility API（辅助功能）

# Linux 专属
# - xdotool：命令行窗口操作
# - xlib：X11 窗口库
# - scrot/gnome-screenshot：截图

# 跨平台策略
# 核心操作用 pyautogui（全平台）
# 平台专属功能用 if/else 分支
# 图像识别用 OpenCV（全平台）

Mac 辅助功能权限

Mac 上 PC 自动化最大的坑：「没权限」。脚本能跑，但「点不动」。必须在系统设置里授权：

Mac 权限配置
# 1. 系统设置 → 隐私与安全性 → 辅助功能
#    添加你的终端（Terminal/iTerm）或 Python 解释器
#    勾选"允许此应用控制您的电脑"

# 2. 自动化权限（控制其他应用）
# 系统设置 → 隐私与安全性 → 自动化
# 勾选"允许控制以下应用"（比如 Excel、Finder）

# 3. 屏幕录制权限（截图需要）
# 系统设置 → 隐私与安全性 → 屏幕录制
# 勾选你的终端

# 4. 验证权限（脚本里）
import subprocess
import platform

def check_mac_permissions():
    """检查 Mac 权限（辅助功能 + 屏幕录制）。"""
    if platform.system() != "Darwin":
        return True   # 非 Mac 跳过

    # 辅助功能权限（osascript 测试）
    result = subprocess.run(
        ["osascript", "-e",
         "tell application \"System Events\" to get name of first process"],
        capture_output=True, text=True
    )
    if result.returncode != 0:
        print("⚠️  辅助功能权限未授权")
        print("   到 系统设置 → 隐私与安全性 → 辅助功能 添加你的终端")
        return False

    # 屏幕录制权限（截图测试）
    import pyautogui
    try:
        screenshot = pyautogui.screenshot()
        assert screenshot.size[0] > 0, "截图全黑（权限不足）"
    except Exception as e:
        print(f"⚠️  屏幕录制权限未授权：{e}")
        return False

    print("✅ Mac 权限检查通过")
    return True

# 脚本开头检查权限
if not check_mac_permissions():
    print("请授权后重新运行脚本")
    exit(1)

Mac 上 PC 自动化「三大权限」：辅助功能（控制鼠标键盘）、自动化（控制其他应用）、屏幕录制（截图）。任何一个没授权，脚本「能跑但不动」。先在系统设置里授权，再跑脚本。

Linux 适配

Linux 没有「控件树」（桌面环境多样：GNOME/KDE/XFCE），主要靠图像识别 + xdotool：

python — Linux 适配
import platform
import subprocess

def start_app_linux(app_command):
    """Linux 启动应用。"""
    subprocess.Popen(app_command, shell=True)

def start_app_cross_platform(app_name):
    """跨平台启动应用。"""
    system = platform.system()
    if system == "Windows":
        subprocess.Popen([f"{app_name}.exe"])
    elif system == "Darwin":   # macOS
        subprocess.Popen(["open", "-a", app_name])
    elif system == "Linux":
        # xdg-open（按文件类型打开）
        # 或具体命令（比如 firefox）
        subprocess.Popen([app_name], shell=True)

def window_op_cross_platform(action, window_title):
    """跨平台窗口操作（最小化/最大化/关闭）。"""
    system = platform.system()
    if system == "Windows":
        import uiautomation as auto
        window = auto.WindowControl(searchDepth=1, subTitle=window_title)
        if action == "minimize":
            window.minimize()
        elif action == "close":
            window.close()
    elif system == "Darwin":
        # AppleScript（Mac）
        script = f'tell application "System Events" to set miniaturized of window 1 of process "{window_title}" to true'
        subprocess.run(["osascript", "-e", script])
    elif system == "Linux":
        # xdotool（Linux）
        if action == "close":
            subprocess.run(["xdotool", "search", "--name", window_title, "windowclose"])
        elif action == "minimize":
            subprocess.run(["xdotool", "search", "--name", window_title, "windowminimize"])

案例：跨平台「打开浏览器 + 截图」测试

把跨平台适配 + 图像识别串起来，写一个「打开浏览器 + 截图 + 验证」的完整案例：

python — 完整案例
import platform
import pyautogui
import cv2
import numpy as np
import subprocess
import time
import os
from datetime import datetime

CURRENT_PLATFORM = platform.system()

class CrossPlatformBrowserTest:
    """跨平台浏览器测试（打开 + 截图 + 验证）"""

    def start_browser(self):
        """启动浏览器（三平台）。"""
        if CURRENT_PLATFORM == "Windows":
            subprocess.Popen(["chrome.exe", "https://example.com"])
        elif CURRENT_PLATFORM == "Darwin":
            subprocess.Popen(["open", "-a", "Google Chrome", "https://example.com"])
        else:   # Linux
            subprocess.Popen(["google-chrome", "https://example.com"])
        time.sleep(5)   # 等浏览器启动

    def find_page_element(self, target_path, confidence=0.9):
        """找页面元素（OpenCV 图像识别，全平台通用）。"""
        screenshot = pyautogui.screenshot()
        screen_img = cv2.cvtColor(np.array(screenshot), cv2.COLOR_RGB2BGR)
        target_img = cv2.imread(target_path)

        if target_img is None:
            return None

        screen_gray = cv2.cvtColor(screen_img, cv2.COLOR_BGR2GRAY)
        target_gray = cv2.cvtColor(target_img, cv2.COLOR_BGR2GRAY)

        result = cv2.matchTemplate(screen_gray, target_gray, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, max_loc = cv2.minMaxLoc(result)

        if max_val >= confidence:
            th, tw = target_gray.shape[:2]
            return (max_loc[0] + tw // 2, max_loc[1] + th // 2, max_val)
        return None

    def take_screenshot(self, name):
        """截图（全平台通用）。"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        path = os.path.join("screenshots", f"{name}_{timestamp}.png")
        os.makedirs(os.path.dirname(path), exist_ok=True)
        pyautogui.screenshot(path)
        print(f"截图：{path}")
        return path

    def test_open_and_verify(self):
        """打开浏览器 + 验证页面。"""
        print(f"平台：{CURRENT_PLATFORM}")

        # 1. 启动浏览器
        self.start_browser()

        # 2. 等页面加载（固定等待 3s）
        time.sleep(3)

        # 3. 验证"Example Domain"标题（图像识别）
        location = self.find_page_element("images/example_domain_title.png", confidence=0.9)
        if location:
            print(f"✅ 找到页面标题：{location[:2]}")
        else:
            print("⚠️ 没找到页面标题（可能加载慢）")

        # 4. 验证"Learn more"链接
        location = self.find_page_element("images/learn_more_link.png", confidence=0.85)
        assert location, "Learn more 链接没找到"
        print(f"✅ 找到 Learn more 链接：{location[:2]}")

        # 5. 点击（全平台通用 pyautogui）
        pyautogui.click(location[0], location[1])
        time.sleep(3)

        # 6. 截图（验证跳转）
        self.take_screenshot("after_click")

        print("\n=== 跨平台浏览器测试完成 ===")

# 运行
if __name__ == "__main__":
    test = CrossPlatformBrowserTest()
    test.test_open_and_verify()

跨平台 PC 自动化「三层架构」：① 平台检测（platform.system()）；② 平台适配层（封装启动应用/窗口操作差异）；③ 测试用例层（平台无关，用 pyautogui + OpenCV）。核心原则：通用操作用 pyautogui/OpenCV，平台差异用 if/else 封装，测试用例「不关心平台」。
